  1119→/** 白天某小时的光强/温度归一化因子（0=日出/日落，1=峰值）
  1120→ *  支持对称（DAY_PEAK=null）与分段非对称（DAY_PEAK=实测峰值时刻）。 */
  1121→function dayShape(h, D) {
  1122→  const sunrise = CONFIG.SUNRISE;
  1123→  const sunset = sunrise + D;
  1124→  const peak = (CONFIG.DAY_PEAK == null) ? (sunrise + D / 2) : CONFIG.DAY_PEAK;
  1125→  const morn = Math.max(1e-6, peak - sunrise);
  1126→  const eve  = Math.max(1e-6, sunset - peak);
  1127→  if (h <= peak) {
  1128→    return Math.sin(Math.PI / 2 * (h - sunrise) / morn);
  1129→  }
  1130→  return Math.sin(Math.PI / 2 * (1 - (h - peak) / eve));
  1131→}

  1133→/** 生成逐时气温曲线（白天分段正弦，夜间恒 Tmin） */
  1134→function generateHourlyTemps(Tmax, Tmin, D) {
  1135→  const points = [];
  1136→  for (let h = 0; h < 24; h += CONFIG.HOUR_STEP) {
  1137→    if (h < CONFIG.SUNRISE || h > CONFIG.SUNRISE + D) {
  1138→      points.push(Tmin);
  1139→    } else {
  1140→      const s = Math.max(0, Math.min(1, dayShape(h, D)));
  1141→      points.push(Tmin + (Tmax - Tmin) * s);
  1142→    }
  1143→  }
  1144→  return points;
  1145→}

  1156→/** 计算有害积热（二次方模型 × 湿度放大系数 × HOUR_STEP 保持积分守恒） */
  1157→function calcHHA(temps, T0, RH = 60) {
  1158→  const humidFactor = 0.5 + (RH / 100);
  1159→  let sum = 0;
  1160→  for (const t of temps) {
  1161→    if (t > T0) sum += Math.pow(t - T0, 2) * humidFactor * CONFIG.HOUR_STEP;
  1162→  }
  1163→  return sum;
  1164→}

  1166→/** 三段线性模型计算产量损失率 */
  1167→function calcLossRate(HHA) {
  1168→  const { MILD, MODERATE, SEVERE, SEVERE_FULL } = CONFIG.HHA_THRESHOLDS;
  1169→  const cM   = CONFIG.LOSS_CAPS.MILD;
  1170→  const cMod = CONFIG.LOSS_CAPS.MODERATE;
  1171→  const cS   = CONFIG.LOSS_CAPS.SEVERE;
  1172→
  1173→  if (HHA < MILD)                                      return 0;
  1174→  if (HHA < MODERATE)                                 return cM * (HHA - MILD) / (MODERATE - MILD);
  1175→  if (HHA < SEVERE)                                   return cM + (cMod - cM) * (HHA - MODERATE) / (SEVERE - MODERATE);
  1176→  if (HHA < SEVERE_FULL)                              return cMod + (cS - cMod) * (HHA - SEVERE) / (SEVERE_FULL - SEVERE);
  1177→  return cS;
  1178→}

  1184→/**
  1185→ * 光合核心：给定有效 PAR 透射率 tau，计算归一化光合量（HOUR_STEP 守恒）
  1186→ */
  1187→function photosynthCore(tau, LSP, LCP, Imax, D) {
  1188→  const Iin = Imax * tau;
  1189→  let total = 0;
  1190→  const Km = (LSP - LCP) * (1 - CONFIG.P_AT_LSP) / CONFIG.P_AT_LSP;
  1191→  for (let h = CONFIG.SUNRISE; h < CONFIG.SUNRISE + D - 1e-6; h += CONFIG.HOUR_STEP) {
  1192→    const s = Math.max(0, Math.min(1, dayShape(h, D)));
  1193→    const I = Iin * s;
  1194→    if (I <= LCP) continue;
  1195→    if (I >= LSP) { total += 1 * CONFIG.HOUR_STEP; continue; }
  1196→    total += (I - LCP) / (I - LCP + Km) * CONFIG.HOUR_STEP;
  1197→  }
  1198→  return total;
  1199→}

  1211→/**
  1212→ * 按 xField 字段线性插值 yField（二分查找）
  1213→ * @warning 调用方必须保证 table 已按 xField 升序排列
  1214→ */
  1215→function linearInterpolate(table, xField, yField, xVal) {
  1216→  const n = table.length;
  1217→  if (n === 0) return NaN;
  1218→  if (n === 1) return table[0][yField];
  1219→
  1220→  // 下界线性外推
  1221→  if (xVal < table[0][xField]) {
  1222→    const dx = table[1][xField] - table[0][xField];
  1223→    if (Math.abs(dx) < 1e-9) return table[0][yField];
  1224→    const slope = (table[1][yField] - table[0][yField]) / dx;
  1225→    return table[0][yField] + slope * (xVal - table[0][xField]);
  1226→  }
  1227→  // 上界线性外推
  1228→  if (xVal > table[n - 1][xField]) {
  1229→    const dx = table[n - 1][xField] - table[n - 2][xField];
  1230→    if (Math.abs(dx) < 1e-9) return table[n - 1][yField];
  1231→    const slope = (table[n - 1][yField] - table[n - 2][yField]) / dx;
  1232→    return table[n - 1][yField] + slope * (xVal - table[n - 1][xField]);
  1233→  }
  1234→
  1235→  // 内部二分查找插值
  1236→  let lo = 0, hi = n - 1;
  1237→  while (lo <= hi) {
  1238→    const mid = (lo + hi) >> 1;
  1239→    if (table[mid][xField] === xVal) return table[mid][yField];
  1240→    else if (table[mid][xField] < xVal) lo = mid + 1;
  1241→    else hi = mid - 1;
  1242→  }
  1243→  const i = hi;
  1244→  const denom = table[i + 1][xField] - table[i][xField];
  1245→  if (Math.abs(denom) < 1e-9) return table[i][yField];
  1246→  const f = (xVal - table[i][xField]) / denom;
  1247→  return table[i][yField] + f * (table[i + 1][yField] - table[i][yField]);
  1248→}
  1249→
  1250→/** 字符串字段：找 xField 最近的行返回 yField
  1251→ *  @warning 调用方必须保证 table 已按 xField 升序排列 */
  1252→function nearestStr(table, xField, yField, xVal) {
  1253→  if (table.length === 0) return '';
  1254→  if (xVal <= table[0][xField])              return table[0][yField];
  1255→  if (xVal >= table[table.length - 1][xField]) return table[table.length - 1][yField];
  1256→
  1257→  let best   = table[0];
  1258→  let bestD  = Infinity;
  1259→  for (const row of table) {
  1260→    const d = Math.abs(row[xField] - xVal);
  1261→    if (d < bestD) { bestD = d; best = row; }
  1262→  }
  1263→  return best[yField];
  1264→}

  1270→/**
  1271→ * Fritsch-Carlson 单调保持三次样条插值
  1272→ * 输入控制点 [{x, y}]，输出等间距 10nm 步长插值曲线
  1273→ * @param {Array<{x:number, y:number}>} points — 已排序控制点 (x递增)
  1274→ * @param {number} xmin, xmax — 输出范围 (nm)
  1275→ * @param {number} step — 输出步长 (nm)
  1276→ * @returns {Array<{x:number, y:number}>} — 等间距插值曲线
  1277→ */
  1278→function fritschCarlsonSpline(points, xmin, xmax, step) {
  1279→  const n = points.length;
  1280→  if (n < 2) {
  1281→    const val = (n === 1) ? Math.max(0, Math.min(1, points[0].y)) : 0;
  1282→    const result = [];
  1283→    for (let x = xmin; x <= xmax + 0.001; x += step) {
  1284→      result.push({ x: +x.toFixed(1), y: val });
  1285→    }
  1286→    return result;
  1287→  }
  1288→
  1289→  // 2 点退化：直接用线性插值
  1290→  if (n === 2) {
  1291→    const pts = [...points].sort((a, b) => a.x - b.x);
  1292→    const dx = pts[1].x - pts[0].x;
  1293→    const dy = pts[1].y - pts[0].y;
  1294→    const result = [];
  1295→    for (let x = xmin; x <= xmax + 0.001; x += step) {
  1296→      const t = Math.max(0, Math.min(1, (x - pts[0].x) / Math.max(1e-9, dx)));
  1297→      const y = Math.max(0, Math.min(1, pts[0].y + t * (pts[1].y - pts[0].y)));
  1298→      result.push({ x: +x.toFixed(1), y });
  1299→    }
  1300→    return result;
  1301→  }
  1302→
  1303→  const pts = [...points].sort((a, b) => a.x - b.x);
  1304→
  1305→  const h = new Array(n - 1);
  1306→  const m = new Array(n - 1);
  1307→  for (let i = 0; i < n - 1; i++) {
  1308→    h[i] = pts[i + 1].x - pts[i].x;
  1309→    m[i] = (pts[i + 1].y - pts[i].y) / Math.max(1e-9, h[i]);
  1310→  }
  1311→
  1312→  const d = new Array(n);
  1313→
  1314→  d[0] = ((2 * h[0] + h[1]) * m[0] - h[0] * m[1]) / Math.max(1e-9, h[0] + h[1]);
  1315→  if (d[0] * m[0] <= 0) d[0] = 0;
  1316→
  1317→  d[n - 1] = ((2 * h[n - 2] + h[n - 3]) * m[n - 2] - h[n - 2] * m[n - 3])
  1318→    / Math.max(1e-9, h[n - 2] + h[n - 3]);
  1319→  if (d[n - 1] * m[n - 2] <= 0) d[n - 1] = 0;
  1320→
  1321→  for (let i = 1; i < n - 1; i++) {
  1322→    if (m[i - 1] * m[i] <= 0) {
  1323→      d[i] = 0;
  1324→    } else {
  1325→      const w1 = 2 * h[i] + h[i - 1];
  1326→      const w2 = h[i] + 2 * h[i - 1];
  1327→      d[i] = (w1 + w2) / (w1 / Math.max(1e-9, m[i - 1]) + w2 / Math.max(1e-9, m[i]));
  1328→    }
  1329→  }
  1330→
  1331→  for (let i = 0; i < n; i++) {
  1332→    const maxSlope = (i < n - 1) ? Math.abs(m[i]) : Math.abs(m[n - 2]);
  1333→    if (Math.abs(d[i]) > 3 * maxSlope) {
  1334→      d[i] = 3 * maxSlope * (d[i] > 0 ? 1 : -1);
  1335→    }
  1336→  }
  1337→
  1338→  const result = [];
  1339→  let seg = 0;
  1340→  for (let x = xmin; x <= xmax + 0.001; x += step) {
  1341→    while (seg < n - 2 && x > pts[seg + 1].x + 0.001) seg++;
  1342→    if (seg >= n - 1) seg = n - 2;
  1343→
  1344→    const x0 = pts[seg].x;
  1345→    const x1 = pts[seg + 1].x;
  1346→    const hi = Math.max(1e-9, x1 - x0);
  1347→    const t = Math.max(0, Math.min(1, (x - x0) / hi));
  1348→
  1349→    const h00 = (1 + 2 * t) * (1 - t) * (1 - t);
  1350→    const h10 = t * (1 - t) * (1 - t);
  1351→    const h01 = t * t * (3 - 2 * t);
  1352→    const h11 = t * t * (t - 1);
  1353→
  1354→    let y = h00 * pts[seg].y + h10 * hi * d[seg] + h01 * pts[seg + 1].y + h11 * hi * d[seg + 1];
  1355→    y = Math.max(0, Math.min(1, y));
  1356→    result.push({ x: +x.toFixed(1), y });
  1357→  }
  1358→
  1359→  return result;
  1360→}

  1362→/**
  1363→ * 光谱积分：计算总遮阳率 S_total 与有效 PAR 透过率 Tau_PAR_eff
  1364→ */
  1365→function integrateSpectrum(curve) {
  1366→  const solar = CONFIG.SOLAR_SPECTRUM_DATA;
  1367→  const plant = CONFIG.PLANT_RESPONSE_DEFAULT;
  1368→  const N = 73;
  1369→
  1370→  if (!curve || curve.length < N) {
  1371→    return { S_total: NaN, Tau_PAR_eff: NaN };
  1372→  }
  1373→
  1374→  let sumSolar = 0, sumWeighted = 0, sumPlantWeighted = 0, sumPlantRef = 0;
  1375→  for (let i = 0; i < N; i++) {
  1376→    const r = curve[i].y;
  1377→    const I = solar[i];
  1378→    const p = plant[i];
  1379→    sumSolar += I;
  1380→    sumWeighted += r * I;
  1381→    sumPlantWeighted += r * I * p;
  1382→    sumPlantRef += I * p;
  1383→  }
  1384→
  1385→  const S_total = sumWeighted / Math.max(1e-9, sumSolar);
  1386→  const Rpar_eff = sumPlantWeighted / Math.max(1e-9, sumPlantRef);
  1387→  const Tau_PAR_eff = Math.max(0, 1 - Rpar_eff);
  1388→
  1389→  return { S_total, Tau_PAR_eff };
  1390→}

  1641→/* ---- 主入口 ---- */
  1642→function findOptimalR(params) {
  1643→  const {
  1644→    LSP, LCP, Tmax, Tmin, D, Imax, k, T0, useTable, RH,
  1645→    spectrumMode, spectrumControlPoints
  1646→  } = params;
  1647→  // 优先使用排序后的 productTable，回退到原始
  1648→  const productTable = params.productTableSorted || params.productTable;
  1649→
  1650→  const baseTemps = generateHourlyTemps(Tmax, Tmin, D);
  1651→  const baseA = calcPhotosynth(0, LSP, LCP, Imax, D);
  1652→  if (baseA === 0) {
  1653→    return { results: [], optimum: null, error: '基准光合量为零，请检查 LSP/LCP/Imax 参数' };
  1654→  }
  1655→
  1656→  /* 光谱模式预计算：归一化基础波形 → 积分得单位振幅常数 → 循环内 O(1) */
  1657→  let K_solar = null;  // S_total(R) = R × K_solar
  1658→  let K_par   = null;  // Rpar_loss(R) = R × K_par
  1659→  if (spectrumMode && spectrumControlPoints && spectrumControlPoints.length >= 2) {
  1660→    const baseCurve = fritschCarlsonSpline(spectrumControlPoints, 380, 1100, 10);
  1661→    const peak = Math.max(...baseCurve.map(p => p.y), 0.001);
  1662→    const normCurve = baseCurve.map(p => ({ x: p.x, y: p.y / peak }));
  1663→    const { S_total: Sn, Tau_PAR_eff: Te } = integrateSpectrum(normCurve);
  1664→    K_solar = Sn;      // 单位振幅下的总遮阳率
  1665→    K_par   = 1 - Te;  // 单位振幅下的 PAR 损失率
  1666→  }
  1667→
  1668→  let bestIndex = -1;
  1669→  let bestY     = -Infinity;
  1670→  const results = [];
  1671→  const cooledBuf = new Array(baseTemps.length);  // 复用缓冲区，避免每次 map 创建新数组
  1672→
  1673→  const steps = Math.round((CONFIG.R_MAX - CONFIG.R_MIN) / CONFIG.R_STEP);
  1674→  for (let i = 0; i <= steps; i++) {
  1675→    const R = CONFIG.R_MIN + i * CONFIG.R_STEP;
  1676→    let deltaT, A_rel;
  1677→
  1678→    if (K_solar != null) {
  1679→      // 光谱模式：O(1) 乘法替代循环内样条+积分
  1680→      const S_total = R * K_solar;
  1681→      const Tau_PAR_eff = 1 - R * K_par;
  1682→      deltaT = getTempDrop(S_total, useTable, productTable, k);
  1683→      A_rel = photosynthCore(Tau_PAR_eff, LSP, LCP, Imax, D) / baseA;
  1684→    } else {
  1685→      deltaT = getTempDrop(R, useTable, productTable, k);
  1686→      A_rel = calcPhotosynth(R, LSP, LCP, Imax, D) / baseA;
  1687→    }
  1688→
  1689→    for (let j = 0; j < baseTemps.length; j++) cooledBuf[j] = baseTemps[j] - deltaT;
  1690→    const HHA = calcHHA(cooledBuf, T0, RH);
  1691→    const L   = calcLossRate(HHA);
  1692→    const Y   = Math.max(0, Math.min(1, A_rel * (1 - L)));
  1693→
  1694→    results.push({
  1695→      R:            Math.round(R * 100) / 100,
  1696→      Y,
  1697→      A_rel,
  1698→      L,
  1699→      HHA:          Math.round(HHA * 10) / 10,
  1700→      Tmax_cooled:  +(Tmax - deltaT).toFixed(1),
  1701→      deltaT:       +deltaT.toFixed(1),
  1702→      S_total:      K_solar != null ? +(R * K_solar).toFixed(3) : null
  1703→    });
  1704→
  1705→    if (Y > bestY) {
  1706→      bestY     = Y;
  1707→      bestIndex = results.length - 1;
  1708→    }
  1709→  }
  1710→
  1711→  /* 近优平台区间 */
  1712→  let plateauRMin = null;
  1713→  let plateauRMax = null;
  1714→  if (bestIndex >= 0 && bestY > -Infinity) {
  1715→    const near = results.filter(r => r.Y >= bestY * (1 - CONFIG.PLATEAU_EPSILON));
  1716→    if (near.length > 0) {
  1717→      plateauRMin = Math.min(...near.map(r => r.R));
  1718→      plateauRMax = Math.max(...near.map(r => r.R));
  1719→    }
  1720→  }
  1721→
  1722→  let optimum = null;
  1723→  if (plateauRMin != null) {
  1724→    optimum = results.find(r => Math.abs(r.R - plateauRMin) < 1e-6) || results[bestIndex];
  1725→  } else if (bestIndex >= 0) {
  1726→    optimum = results[bestIndex];
  1727→  }
  1728→
  1729→  return { results, optimum, plateauRMin, plateauRMax, bestIndex, error: null };
  1730→}

  1816→ * K 值模式稀释比建议（含估算 detail）
  1817→ * 基于 standardRatio → coverage 经验映射估算
  1818→ */
  1819→function interpolateDilution(Ropt, mapping, sprayAreaMu) {
  1820→  const muToM2 = (sprayAreaMu || 1) * CONFIG.MU_TO_M2;
  1821→
  1822→  if (!mapping || mapping.length === 0) {
  1823→    return { value: '请填写稀释比映射表', status: 'empty', mode: 'k', detail: null };
  1824→  }
  1825→  if (mapping.length < 2) {
  1826→    return { value: '请至少填写两组映射数据', status: 'insufficient', mode: 'k', detail: null };
  1827→  }
  1828→
  1829→  const sorted = [...mapping].sort((a, b) => a.r - b.r);
  1830→
  1831→  if (Ropt < sorted[0].r || Ropt > sorted[sorted.length - 1].r) {
  1832→    const p = v => (v * 100).toFixed(0) + '%';
  1833→    return {
  1834→      value: `R_opt 超出范围 [${p(sorted[0].r)}–${p(sorted[sorted.length - 1].r)}]`,
  1835→      status: 'out_of_range',
  1836→      mode: 'k',
  1837→      detail: null
  1838→    };
  1839→  }
  1840→
  1841→  let lo = sorted[0];
  1842→  let hi = sorted[sorted.length - 1];
  1843→  for (let i = 0; i < sorted.length - 1; i++) {
  1844→    if (Ropt >= sorted[i].r && Ropt <= sorted[i + 1].r) {
  1845→      lo = sorted[i];
  1846→      hi = sorted[i + 1];
  1847→      break;
  1848→    }
  1849→  }
  1850→
  1851→  const ratio = (Math.abs(hi.r - lo.r) < 1e-9)
  1852→    ? lo.ratio
  1853→    : lo.ratio + (Ropt - lo.r) / (hi.r - lo.r) * (hi.ratio - lo.ratio);
  1854→
  1855→  // 基于 CONFIG.K_MODE_COVERAGE_MAP 二分插值估算 coverage
  1856→  const ratioCovg = CONFIG.K_MODE_COVERAGE_MAP;
  1857→  const ri = Math.max(1, Math.min(20, ratio));
  1858→  const estCoverage = Math.max(50, Math.min(5000, linearInterpolate(ratioCovg, 'r', 'c', ri)));
  1859→
  1860→  const powderKg = muToM2 / estCoverage;
  1861→  const waterL   = powderKg * ratio;
  1862→
  1863→  return {
  1864→    value: `建议兑水比例 1:${ratio.toFixed(1)}`,
  1865→    status: 'ok',
  1866→    mode: 'k',
  1867→    detail: {
  1868→      ratio:    `1:${ratio.toFixed(1)}`,
  1869→      coverage: `${estCoverage.toFixed(0)} ㎡/kg (估算)`,
  1870→      powder:   String(powderKg),
  1871→      water:    String(waterL),
  1872→      powderKg,
  1873→      waterL,
  1874→      isEstimate: true
  1875→    }
  1876→  };
  1877→}

  1885→/** 安全省钱：近优平台 1/4 分位，若无平台则回退 bestIndex */
  1886→function safeSavePoint(results, bestIndex, plateau) {
  1887→  if (plateau && plateau.rMin != null && plateau.rMax != null) {
  1888→    const rQuarter = plateau.rMin + (plateau.rMax - plateau.rMin) * CONFIG.PLATEAU_QUARTER;
  1889→    let nearestIdx = bestIndex, minDiff = Infinity;
  1890→    for (let i = 0; i < results.length; i++) {
  1891→      const diff = Math.abs(results[i].R - rQuarter);
  1892→      if (diff < minDiff) { minDiff = diff; nearestIdx = i; }
  1893→    }
  1894→    return results[nearestIdx];
  1895→  }
  1896→  return results[bestIndex];
  1897→}

  1910→/**

  1912→ * @param {number} bestIndex — 最高 Y 的索引
  1913→ * @param {object} plateau — { rMin, rMax } 或 null

  1915→ * @param {object} enabledFlags — { safe_save: true, max_y: boolean, absolute_temp: boolean }
  1916→ * @returns {{ points: Array, warnings: Array }}

  1918→function extractStrategies(results, bestIndex, plateau, params, enabledFlags) {
  1919→  const points = [];
  1920→  const warnings = [];
  1921→  if (!results || results.length === 0) return { points, warnings };
  1922→
  1923→  // 安全省钱始终生成
  1924→  const baseData = safeSavePoint(results, bestIndex, plateau);
  1925→  points.push(makeStrategyPoint('safe_save', STRATEGY_DEFS.safe_save, baseData, 'preset'));
  1926→
  1927→  if (enabledFlags.max_y) {
  1928→    const maxPt = results[bestIndex];
  1929→    points.push(makeStrategyPoint('max_y', STRATEGY_DEFS.max_y, maxPt, 'preset'));
  1930→  }
  1931→
  1932→  if (enabledFlags.absolute_temp) {
  1933→    const safeResults = results.filter(r => r.Tmax_cooled <= params.T0 + 0.05);
  1934→    if (safeResults.length === 0) {
  1935→      warnings.push({ level: 'warning',
  1936→        text: '当前气候极端，降温剂无法保证棚温降至阈值以下，建议结合遮阳网或通风措施' });
  1937→    } else {
  1938→      // 取温度安全范围内综合效益最优（而非最小R卡边，天气微变就破防）
  1939→      const bestInSafe = safeResults.reduce((a, b) => a.Y > b.Y ? a : b);
  1940→      points.push(makeStrategyPoint('absolute_temp', STRATEGY_DEFS.absolute_temp, bestInSafe, 'preset'));
  1941→    }
  1942→  }
  1943→
  1944→  return { points, warnings };
  1945→}
