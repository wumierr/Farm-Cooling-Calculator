  1213→  1213→ * @warning 调用方必须保证 table 已按 xField 升序排列
  1214→  1214→ */
  1215→  1215→function linearInterpolate(table, xField, yField, xVal) {
  1216→  1216→  const n = table.length;
  1217→  1217→  if (n === 0) return NaN;
  1218→  1218→  if (n === 1) return table[0][yField];
  1219→  1219→
  1220→  1220→  // 下界线性外推
  1221→  1221→  if (xVal < table[0][xField]) {
  1222→  1222→    const dx = table[1][xField] - table[0][xField];
  1223→  1223→    if (Math.abs(dx) < 1e-9) return table[0][yField];
  1224→  1224→    const slope = (table[1][yField] - table[0][yField]) / dx;
  1225→  1225→    return table[0][yField] + slope * (xVal - table[0][xField]);
  1226→  1226→  }
  1227→  1227→  // 上界线性外推
  1228→  1228→  if (xVal > table[n - 1][xField]) {
  1229→  1229→    const dx = table[n - 1][xField] - table[n - 2][xField];
  1230→  1230→    if (Math.abs(dx) < 1e-9) return table[n - 1][yField];
  1231→  1231→    const slope = (table[n - 1][yField] - table[n - 2][yField]) / dx;
  1232→  1232→    return table[n - 1][yField] + slope * (xVal - table[n - 1][xField]);
  1233→  1233→  }
  1234→  1234→
  1235→  1235→  // 内部二分查找插值
  1236→  1236→  let lo = 0, hi = n - 1;
  1237→  1237→  while (lo <= hi) {
  1238→  1238→    const mid = (lo + hi) >> 1;
  1239→  1239→    if (table[mid][xField] === xVal) return table[mid][yField];
  1240→  1240→    else if (table[mid][xField] < xVal) lo = mid + 1;
  1241→  1241→    else hi = mid - 1;
  1242→  1242→  }
  1243→  1243→  const i = hi;
  1244→  1244→  const denom = table[i + 1][xField] - table[i][xField];
  1245→  1245→  if (Math.abs(denom) < 1e-9) return table[i][yField];
  1246→  1246→  const f = (xVal - table[i][xField]) / denom;
  1247→  1247→  return table[i][yField] + f * (table[i + 1][yField] - table[i][yField]);
  1248→  1248→}
  1249→  1249→
  1250→  1250→/** 字符串字段：找 xField 最近的行返回 yField
  1251→  1251→ *  @warning 调用方必须保证 table 已按 xField 升序排列 */
  1252→  1252→function nearestStr(table, xField, yField, xVal) {
  1253→  1253→  if (table.length === 0) return '';
  1254→  1254→  if (xVal <= table[0][xField])              return table[0][yField];
  1255→  1255→  if (xVal >= table[table.length - 1][xField]) return table[table.length - 1][yField];
  1256→  1256→
  1257→  1257→  let best   = table[0];
  1258→  1258→  let bestD  = Infinity;
  1259→  1259→  for (const row of table) {
  1260→  1260→    const d = Math.abs(row[xField] - xVal);
  1261→  1261→    if (d < bestD) { bestD = d; best = row; }
  1262→  1262→  }
  1263→  1263→  return best[yField];
  1264→  1264→}

  1273→  1273→ * @param {Array<{x:number, y:number}>} points — 已排序控制点 (x递增)
  1274→  1274→ * @param {number} xmin, xmax — 输出范围 (nm)
  1275→  1275→ * @param {number} step — 输出步长 (nm)
  1276→  1276→ * @returns {Array<{x:number, y:number}>} — 等间距插值曲线
  1277→  1277→ */
  1278→  1278→function fritschCarlsonSpline(points, xmin, xmax, step) {
  1279→  1279→  const n = points.length;
  1280→  1280→  if (n < 2) {
  1281→  1281→    const val = (n === 1) ? Math.max(0, Math.min(1, points[0].y)) : 0;
  1282→  1282→    const result = [];
  1283→  1283→    for (let x = xmin; x <= xmax + 0.001; x += step) {
  1284→  1284→      result.push({ x: +x.toFixed(1), y: val });
  1285→  1285→    }
  1286→  1286→    return result;
  1287→  1287→  }
  1288→  1288→
  1289→  1289→  // 2 点退化：直接用线性插值
  1290→  1290→  if (n === 2) {
  1291→  1291→    const pts = [...points].sort((a, b) => a.x - b.x);
  1292→  1292→    const dx = pts[1].x - pts[0].x;
  1293→  1293→    const dy = pts[1].y - pts[0].y;
  1294→  1294→    const result = [];
  1295→  1295→    for (let x = xmin; x <= xmax + 0.001; x += step) {
  1296→  1296→      const t = Math.max(0, Math.min(1, (x - pts[0].x) / Math.max(1e-9, dx)));
  1297→  1297→      const y = Math.max(0, Math.min(1, pts[0].y + t * (pts[1].y - pts[0].y)));
  1298→  1298→      result.push({ x: +x.toFixed(1), y });
  1299→  1299→    }
  1300→  1300→    return result;
  1301→  1301→  }
  1302→  1302→
  1303→  1303→  const pts = [...points].sort((a, b) => a.x - b.x);
  1304→  1304→
  1305→  1305→  const h = new Array(n - 1);
  1306→  1306→  const m = new Array(n - 1);
  1307→  1307→  for (let i = 0; i < n - 1; i++) {
  1308→  1308→    h[i] = pts[i + 1].x - pts[i].x;
  1309→  1309→    m[i] = (pts[i + 1].y - pts[i].y) / Math.max(1e-9, h[i]);
  1310→  1310→  }
  1311→  1311→
  1312→  1312→  const d = new Array(n);
  1313→  1313→
  1314→  1314→  d[0] = ((2 * h[0] + h[1]) * m[0] - h[0] * m[1]) / Math.max(1e-9, h[0] + h[1]);
  1315→  1315→  if (d[0] * m[0] <= 0) d[0] = 0;
  1316→  1316→
  1317→  1317→  d[n - 1] = ((2 * h[n - 2] + h[n - 3]) * m[n - 2] - h[n - 2] * m[n - 3])
  1318→  1318→    / Math.max(1e-9, h[n - 2] + h[n - 3]);
  1319→  1319→  if (d[n - 1] * m[n - 2] <= 0) d[n - 1] = 0;
  1320→  1320→
  1321→  1321→  for (let i = 1; i < n - 1; i++) {
  1322→  1322→    if (m[i - 1] * m[i] <= 0) {
  1323→  1323→      d[i] = 0;
  1324→  1324→    } else {
  1325→  1325→      const w1 = 2 * h[i] + h[i - 1];
  1326→  1326→      const w2 = h[i] + 2 * h[i - 1];
  1327→  1327→      d[i] = (w1 + w2) / (w1 / Math.max(1e-9, m[i - 1]) + w2 / Math.max(1e-9, m[i]));
  1328→  1328→    }
  1329→  1329→  }
  1330→  1330→
  1331→  1331→  for (let i = 0; i < n; i++) {
  1332→  1332→    const maxSlope = (i < n - 1) ? Math.abs(m[i]) : Math.abs(m[n - 2]);
  1333→  1333→    if (Math.abs(d[i]) > 3 * maxSlope) {
  1334→  1334→      d[i] = 3 * maxSlope * (d[i] > 0 ? 1 : -1);
  1335→  1335→    }
  1336→  1336→  }
  1337→  1337→
  1338→  1338→  const result = [];
  1339→  1339→  let seg = 0;
  1340→  1340→  for (let x = xmin; x <= xmax + 0.001; x += step) {
  1341→  1341→    while (seg < n - 2 && x > pts[seg + 1].x + 0.001) seg++;
  1342→  1342→    if (seg >= n - 1) seg = n - 2;
  1343→  1343→
  1344→  1344→    const x0 = pts[seg].x;
  1345→  1345→    const x1 = pts[seg + 1].x;
  1346→  1346→    const hi = Math.max(1e-9, x1 - x0);
  1347→  1347→    const t = Math.max(0, Math.min(1, (x - x0) / hi));
  1348→  1348→
  1349→  1349→    const h00 = (1 + 2 * t) * (1 - t) * (1 - t);
  1350→  1350→    const h10 = t * (1 - t) * (1 - t);
  1351→  1351→    const h01 = t * t * (3 - 2 * t);
  1352→  1352→    const h11 = t * t * (t - 1);
  1353→  1353→
  1354→  1354→    let y = h00 * pts[seg].y + h10 * hi * d[seg] + h01 * pts[seg + 1].y + h11 * hi * d[seg + 1];
  1355→  1355→    y = Math.max(0, Math.min(1, y));
  1356→  1356→    result.push({ x: +x.toFixed(1), y });
  1357→  1357→  }
  1358→  1358→
  1359→  1359→  return result;
  1360→  1360→}
  1361→  1361→
  1362→  1362→/**
  1363→  1363→ * 光谱积分：计算总遮阳率 S_total 与有效 PAR 透过率 Tau_PAR_eff
  1364→  1364→ */
  1365→  1365→function integrateSpectrum(curve) {
  1366→  1366→  const solar = CONFIG.SOLAR_SPECTRUM_DATA;
  1367→  1367→  const plant = CONFIG.PLANT_RESPONSE_DEFAULT;
  1368→  1368→  const N = 73;
  1369→  1369→
  1370→  1370→  if (!curve || curve.length < N) {
  1371→  1371→    return { S_total: NaN, Tau_PAR_eff: NaN };
  1372→  1372→  }
  1373→  1373→
  1374→  1374→  let sumSolar = 0, sumWeighted = 0, sumPlantWeighted = 0, sumPlantRef = 0;
  1375→  1375→  for (let i = 0; i < N; i++) {
  1376→  1376→    const r = curve[i].y;
  1377→  1377→    const I = solar[i];
  1378→  1378→    const p = plant[i];
  1379→  1379→    sumSolar += I;
  1380→  1380→    sumWeighted += r * I;
  1381→  1381→    sumPlantWeighted += r * I * p;
  1382→  1382→    sumPlantRef += I * p;
  1383→  1383→  }
  1384→  1384→
  1385→  1385→  const S_total = sumWeighted / Math.max(1e-9, sumSolar);
  1386→  1386→  const Rpar_eff = sumPlantWeighted / Math.max(1e-9, sumPlantRef);
  1387→  1387→  const Tau_PAR_eff = Math.max(0, 1 - Rpar_eff);
  1388→  1388→
  1389→  1389→  return { S_total, Tau_PAR_eff };
  1390→  1390→}
