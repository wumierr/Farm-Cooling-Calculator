  1700→      Tmax_cooled:  +(Tmax - deltaT).toFixed(1),
  1701→      deltaT:       +deltaT.toFixed(1),
  1702→      S_total:      K_solar != null ? +(R * K_solar).toFixed(3) : null
  1703→    });
  1704→
  1705→    if (Y > bestY) {
  1706→      bestY     = Y;
  1707→      bestIndex = results.length - 1;
  1708→    }
  1709→  }
  1710→
  1711→  /* 近优平台区间 */
  1712→  let plateauRMin = null;
  1713→  let plateauRMax = null;
  1714→  if (bestIndex >= 0 && bestY > -Infinity) {
  1715→    const near = results.filter(r => r.Y >= bestY * (1 - CONFIG.PLATEAU_EPSILON));
  1716→    if (near.length > 0) {
  1717→      plateauRMin = Math.min(...near.map(r => r.R));
  1718→      plateauRMax = Math.max(...near.map(r => r.R));
  1719→    }
  1720→  }
  1721→
  1722→  let optimum = null;
  1723→  if (plateauRMin != null) {
  1724→    optimum = results.find(r => Math.abs(r.R - plateauRMin) < 1e-6) || results[bestIndex];
  1725→  } else if (bestIndex >= 0) {
  1726→    optimum = results[bestIndex];
  1727→  }
  1728→
  1729→  return { results, optimum, plateauRMin, plateauRMax, bestIndex, error: null };
  1730→}

  1815→/**
  1816→ * K 值模式稀释比建议（含估算 detail）
  1817→ * 基于 standardRatio → coverage 经验映射估算
  1818→ */
  1819→function interpolateDilution(Ropt, mapping, sprayAreaMu) {
  1820→  const muToM2 = (sprayAreaMu || 1) * CONFIG.MU_TO_M2;
  1821→
  1822→  if (!mapping || mapping.length === 0) {
  1823→    return { value: '请填写稀释比映射表', status: 'empty', mode: 'k', detail: null };
  1824→  }
  1825→  if (mapping.length < 2) {
  1826→    return { value: '请至少填写两组映射数据', status: 'insufficient', mode: 'k', detail: null };
  1827→  }
  1828→
  1829→  const sorted = [...mapping].sort((a, b) => a.r - b.r);
  1830→
  1831→  if (Ropt < sorted[0].r || Ropt > sorted[sorted.length - 1].r) {
  1832→    const p = v => (v * 100).toFixed(0) + '%';
  1833→    return {
  1834→      value: `R_opt 超出范围 [${p(sorted[0].r)}–${p(sorted[sorted.length - 1].r)}]`,
  1835→      status: 'out_of_range',
  1836→      mode: 'k',
  1837→      detail: null
  1838→    };
  1839→  }
  1840→
  1841→  let lo = sorted[0];
  1842→  let hi = sorted[sorted.length - 1];
  1843→  for (let i = 0; i < sorted.length - 1; i++) {
  1844→    if (Ropt >= sorted[i].r && Ropt <= sorted[i + 1].r) {
  1845→      lo = sorted[i];
  1846→      hi = sorted[i + 1];
  1847→      break;
  1848→    }
  1849→  }
  1850→
  1851→  const ratio = (Math.abs(hi.r - lo.r) < 1e-9)
  1852→    ? lo.ratio
  1853→    : lo.ratio + (Ropt - lo.r) / (hi.r - lo.r) * (hi.ratio - lo.ratio);
  1854→
  1855→  // 基于 CONFIG.K_MODE_COVERAGE_MAP 二分插值估算 coverage
  1856→  const ratioCovg = CONFIG.K_MODE_COVERAGE_MAP;
  1857→  const ri = Math.max(1, Math.min(20, ratio));
  1858→  const estCoverage = Math.max(50, Math.min(5000, linearInterpolate(ratioCovg, 'r', 'c', ri)));
  1859→
  1860→  const powderKg = muToM2 / estCoverage;
  1861→  const waterL   = powderKg * ratio;
  1862→
  1863→  return {
  1864→    value: `建议兑水比例 1:${ratio.toFixed(1)}`,
  1865→    status: 'ok',
  1866→    mode: 'k',
  1867→    detail: {
  1868→      ratio:    `1:${ratio.toFixed(1)}`,
  1869→      coverage: `${estCoverage.toFixed(0)} ㎡/kg (估算)`,
  1870→      powder:   String(powderKg),
  1871→      water:    String(waterL),
  1872→      powderKg,
  1873→      waterL,
  1874→      isEstimate: true
  1875→    }
  1876→  };
  1877→}

  1885→/** 安全省钱：近优平台 1/4 分位，若无平台则回退 bestIndex */
  1886→function safeSavePoint(results, bestIndex, plateau) {
  1887→  if (plateau && plateau.rMin != null && plateau.rMax != null) {
  1888→    const rQuarter = plateau.rMin + (plateau.rMax - plateau.rMin) * CONFIG.PLATEAU_QUARTER;
  1889→    let nearestIdx = bestIndex, minDiff = Infinity;
  1890→    for (let i = 0; i < results.length; i++) {
  1891→      const diff = Math.abs(results[i].R - rQuarter);
  1892→      if (diff < minDiff) { minDiff = diff; nearestIdx = i; }
  1893→    }
  1894→    return results[nearestIdx];
  1895→  }
  1896→  return results[bestIndex];
  1897→}

  1918→function extractStrategies(results, bestIndex, plateau, params, enabledFlags) {
  1919→  const points = [];
  1920→  const warnings = [];
  1921→  if (!results || results.length === 0) return { points, warnings };
  1922→

  1924→  const baseData = safeSavePoint(results, bestIndex, plateau);

  1926→
  1927→  if (enabledFlags.max_y) {
  1928→    const maxPt = results[bestIndex];
  1929→    points.push(makeStrategyPoint('max_y', STRATEGY_DEFS.max_y, maxPt, 'preset'));
  1930→  }
  1931→
  1932→  if (enabledFlags.absolute_temp) {
  1933→    const safeResults = results.filter(r => r.Tmax_cooled <= params.T0 + 0.05);
  1934→    if (safeResults.length === 0) {
  1935→      warnings.push({ level: 'warning',
  1936→        text: '当前气候极端，降温剂无法保证棚温降至阈值以下，建议结合遮阳网或通风措施' });
  1937→    } else {
  1938→      // 取温度安全范围内综合效益最优（而非最小R卡边，天气微变就破防）
  1939→      const bestInSafe = safeResults.reduce((a, b) => a.Y > b.Y ? a : b);
  1940→      points.push(makeStrategyPoint('absolute_temp', STRATEGY_DEFS.absolute_temp, bestInSafe, 'preset'));
  1941→    }
  1942→  }
  1943→
  1944→  return { points, warnings };
  1945→}
