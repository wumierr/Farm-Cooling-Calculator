/** 白天某小时的光强/温度归一化因子（0=日出/日落，1=峰值）
 *  支持对称（DAY_PEAK=null）与分段非对称（DAY_PEAK=实测峰值时刻）。 */
function dayShape(h, D) {
  const sunrise = CONFIG.SUNRISE;
  const sunset = sunrise + D;
  const peak = (CONFIG.DAY_PEAK == null) ? (sunrise + D / 2) : CONFIG.DAY_PEAK;
  const morn = Math.max(1e-6, peak - sunrise);
  const eve  = Math.max(1e-6, sunset - peak);
  if (h <= peak) {
    return Math.sin(Math.PI / 2 * (h - sunrise) / morn);
  }
  return Math.sin(Math.PI / 2 * (1 - (h - peak) / eve));
}

/** 生成逐时气温曲线（白天分段正弦，夜间恒 Tmin） */
function generateHourlyTemps(Tmax, Tmin, D) {
  const points = [];
  for (let h = 0; h < 24; h += CONFIG.HOUR_STEP) {
    if (h < CONFIG.SUNRISE || h > CONFIG.SUNRISE + D) {
      points.push(Tmin);
    } else {
      const s = Math.max(0, Math.min(1, dayShape(h, D)));
      points.push(Tmin + (Tmax - Tmin) * s);
    }
  }
  return points;
}

/** 计算有害积热（二次方模型 × 湿度放大系数 × HOUR_STEP 保持积分守恒） */
function calcHHA(temps, T0, RH = 60) {
  const humidFactor = 0.5 + (RH / 100);
  let sum = 0;
  for (const t of temps) {
    if (t > T0) sum += Math.pow(t - T0, 2) * humidFactor * CONFIG.HOUR_STEP;
  }
  return sum;
}

/** 三段线性模型计算产量损失率 */
function calcLossRate(HHA) {
  const { MILD, MODERATE, SEVERE, SEVERE_FULL } = CONFIG.HHA_THRESHOLDS;
  const cM   = CONFIG.LOSS_CAPS.MILD;
  const cMod = CONFIG.LOSS_CAPS.MODERATE;
  const cS   = CONFIG.LOSS_CAPS.SEVERE;

  if (HHA < MILD)                                      return 0;
  if (HHA < MODERATE)                                 return cM * (HHA - MILD) / (MODERATE - MILD);
  if (HHA < SEVERE)                                   return cM + (cMod - cM) * (HHA - MODERATE) / (SEVERE - MODERATE);
  if (HHA < SEVERE_FULL)                              return cMod + (cS - cMod) * (HHA - SEVERE) / (SEVERE_FULL - SEVERE);
  return cS;
}

/**
 * 光合核心：给定有效 PAR 透射率 tau，计算归一化光合量（HOUR_STEP 守恒）
 */
function photosynthCore(tau, LSP, LCP, Imax, D) {
  const Iin = Imax * tau;
  let total = 0;
  const Km = (LSP - LCP) * (1 - CONFIG.P_AT_LSP) / CONFIG.P_AT_LSP;
  for (let h = CONFIG.SUNRISE; h < CONFIG.SUNRISE + D - 1e-6; h += CONFIG.HOUR_STEP) {
    const s = Math.max(0, Math.min(1, dayShape(h, D)));
    const I = Iin * s;
    if (I <= LCP) continue;
    if (I >= LSP) { total += 1 * CONFIG.HOUR_STEP; continue; }
    total += (I - LCP) / (I - LCP + Km) * CONFIG.HOUR_STEP;
  }
  return total;
}

/** 全遮阳率模式：R 为标量遮阳率，通过 PAR_TRANSMISSION_FACTOR 折算 */
function calcPhotosynth(R, LSP, LCP, Imax, D) {
  const tau = Math.max(0, 1 - R * CONFIG.PAR_TRANSMISSION_FACTOR);
  return photosynthCore(tau, LSP, LCP, Imax, D);
}

/**
 * 按 xField 字段线性插值 yField（二分查找）
 * @warning 调用方必须保证 table 已按 xField 升序排列
 */
function linearInterpolate(table, xField, yField, xVal) {
  const n = table.length;
  if (n === 0) return NaN;
  if (n === 1) return table[0][yField];

  // 下界线性外推
  if (xVal < table[0][xField]) {
    const dx = table[1][xField] - table[0][xField];
    if (Math.abs(dx) < 1e-9) return table[0][yField];
    const slope = (table[1][yField] - table[0][yField]) / dx;
    return table[0][yField] + slope * (xVal - table[0][xField]);
  }
  // 上界线性外推
  if (xVal > table[n - 1][xField]) {
    const dx = table[n - 1][xField] - table[n - 2][xField];
    if (Math.abs(dx) < 1e-9) return table[n - 1][yField];
    const slope = (table[n - 1][yField] - table[n - 2][yField]) / dx;
    return table[n - 1][yField] + slope * (xVal - table[n - 1][xField]);
  }

  // 内部二分查找插值
  let lo = 0, hi = n - 1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    if (table[mid][xField] === xVal) return table[mid][yField];
    else if (table[mid][xField] < xVal) lo = mid + 1;
    else hi = mid - 1;
  }
  const i = hi;
  const denom = table[i + 1][xField] - table[i][xField];
  if (Math.abs(denom) < 1e-9) return table[i][yField];
  const f = (xVal - table[i][xField]) / denom;
  return table[i][yField] + f * (table[i + 1][yField] - table[i][yField]);
}

/** 字符串字段：找 xField 最近的行返回 yField
 *  @warning 调用方必须保证 table 已按 xField 升序排列 */
function nearestStr(table, xField, yField, xVal) {
  if (table.length === 0) return '';
  if (xVal <= table[0][xField])              return table[0][yField];
  if (xVal >= table[table.length - 1][xField]) return table[table.length - 1][yField];

  let best   = table[0];
  let bestD  = Infinity;
  for (const row of table) {
    const d = Math.abs(row[xField] - xVal);
    if (d < bestD) { bestD = d; best = row; }
  }
  return best[yField];
}

/**
 * Fritsch-Carlson 单调保持三次样条插值
 * 输入控制点 [{x, y}]，输出等间距 10nm 步长插值曲线
 * @param {Array<{x:number, y:number}>} points — 已排序控制点 (x递增)
 * @param {number} xmin, xmax — 输出范围 (nm)
 * @param {number} step — 输出步长 (nm)
 * @returns {Array<{x:number, y:number}>} — 等间距插值曲线
 */
function fritschCarlsonSpline(points, xmin, xmax, step) {
  const n = points.length;
  if (n < 2) {
    const val = (n === 1) ? Math.max(0, Math.min(1, points[0].y)) : 0;
    const result = [];
    for (let x = xmin; x <= xmax + 0.001; x += step) {
      result.push({ x: +x.toFixed(1), y: val });
    }
    return result;
  }

  // 2 点退化：直接用线性插值
  if (n === 2) {
    const pts = [...points].sort((a, b) => a.x - b.x);
    const dx = pts[1].x - pts[0].x;
    const dy = pts[1].y - pts[0].y;
    const result = [];
    for (let x = xmin; x <= xmax + 0.001; x += step) {
      const t = Math.max(0, Math.min(1, (x - pts[0].x) / Math.max(1e-9, dx)));
      const y = Math.max(0, Math.min(1, pts[0].y + t * (pts[1].y - pts[0].y)));
      result.push({ x: +x.toFixed(1), y });
    }
    return result;
  }

  const pts = [...points].sort((a, b) => a.x - b.x);

  const h = new Array(n - 1);
  const m = new Array(n - 1);
  for (let i = 0; i < n - 1; i++) {
    h[i] = pts[i + 1].x - pts[i].x;
    m[i] = (pts[i + 1].y - pts[i].y) / Math.max(1e-9, h[i]);
  }

  const d = new Array(n);

  d[0] = ((2 * h[0] + h[1]) * m[0] - h[0] * m[1]) / Math.max(1e-9, h[0] + h[1]);
  if (d[0] * m[0] <= 0) d[0] = 0;

  d[n - 1] = ((2 * h[n - 2] + h[n - 3]) * m[n - 2] - h[n - 2] * m[n - 3])
    / Math.max(1e-9, h[n - 2] + h[n - 3]);
  if (d[n - 1] * m[n - 2] <= 0) d[n - 1] = 0;

  for (let i = 1; i < n - 1; i++) {
    if (m[i - 1] * m[i] <= 0) {
      d[i] = 0;
    } else {
      const w1 = 2 * h[i] + h[i - 1];
      const w2 = h[i] + 2 * h[i - 1];
      d[i] = (w1 + w2) / (w1 / Math.max(1e-9, m[i - 1]) + w2 / Math.max(1e-9, m[i]));
    }
  }

  for (let i = 0; i < n; i++) {
    const maxSlope = (i < n - 1) ? Math.abs(m[i]) : Math.abs(m[n - 2]);
    if (Math.abs(d[i]) > 3 * maxSlope) {
      d[i] = 3 * maxSlope * (d[i] > 0 ? 1 : -1);
    }
  }

  const result = [];
  let seg = 0;
  for (let x = xmin; x <= xmax + 0.001; x += step) {
    while (seg < n - 2 && x > pts[seg + 1].x + 0.001) seg++;
    if (seg >= n - 1) seg = n - 2;

    const x0 = pts[seg].x;
    const x1 = pts[seg + 1].x;
    const hi = Math.max(1e-9, x1 - x0);
    const t = Math.max(0, Math.min(1, (x - x0) / hi));

    const h00 = (1 + 2 * t) * (1 - t) * (1 - t);
    const h10 = t * (1 - t) * (1 - t);
    const h01 = t * t * (3 - 2 * t);
    const h11 = t * t * (t - 1);

    let y = h00 * pts[seg].y + h10 * hi * d[seg] + h01 * pts[seg + 1].y + h11 * hi * d[seg + 1];
    y = Math.max(0, Math.min(1, y));
    result.push({ x: +x.toFixed(1), y });
  }

  return result;
}

/**
 * 光谱积分：计算总遮阳率 S_total 与有效 PAR 透过率 Tau_PAR_eff
 */
function integrateSpectrum(curve) {
  const solar = CONFIG.SOLAR_SPECTRUM_DATA;
  const plant = CONFIG.PLANT_RESPONSE_DEFAULT;
  const N = 73;

  if (!curve || curve.length < N) {
    return { S_total: NaN, Tau_PAR_eff: NaN };
  }

  let sumSolar = 0, sumWeighted = 0, sumPlantWeighted = 0, sumPlantRef = 0;
  for (let i = 0; i < N; i++) {
    const r = curve[i].y;
    const I = solar[i];
    const p = plant[i];
    sumSolar += I;
    sumWeighted += r * I;
    sumPlantWeighted += r * I * p;
    sumPlantRef += I * p;
  }

  const S_total = sumWeighted / Math.max(1e-9, sumSolar);
  const Rpar_eff = sumPlantWeighted / Math.max(1e-9, sumPlantRef);
  const Tau_PAR_eff = Math.max(0, 1 - Rpar_eff);

  return { S_total, Tau_PAR_eff };
}

/* ---- Validation ---- */
function validateParams(params) {
  const errors = [];

  /* LSP */
  if (isNaN(params.LSP))           errors.push('光饱和点 (LSP) 必须为数字');
  else if (params.LSP < 500 || params.LSP > 2500) errors.push('光饱和点 (LSP) 应在 500-2500 之间');

  /* LCP */
  if (isNaN(params.LCP))           errors.push('光补偿点 (LCP) 必须为数字');
  else if (params.LCP < 1 || params.LCP > 500)    errors.push('光补偿点 (LCP) 应在 1-500 之间');

  /* LSP > LCP */
  if (!isNaN(params.LSP) && !isNaN(params.LCP) && params.LSP <= params.LCP)
    errors.push('光饱和点 (LSP) 必须大于光补偿点 (LCP)');

  /* Tmax */
  if (isNaN(params.Tmax))          errors.push('日最高温 (Tmax) 必须为数字');
  else if (params.Tmax < 20 || params.Tmax > 60)  errors.push('日最高温 (Tmax) 应在 20-60°C 之间');

  /* Tmin */
  if (isNaN(params.Tmin))          errors.push('日最低温 (Tmin) 必须为数字');
  else if (params.Tmin < 5 || params.Tmin > 40)   errors.push('日最低温 (Tmin) 应在 5-40°C 之间');

  /* Tmax > Tmin */
  if (!isNaN(params.Tmax) && !isNaN(params.Tmin) && params.Tmax <= params.Tmin)
    errors.push('日最高温 (Tmax) 必须大于日最低温 (Tmin)');

  /* D */
  if (isNaN(params.D))             errors.push('日长 (D) 必须为数字');
  else if (params.D < 8 || params.D > 16)          errors.push('日长 (D) 应在 8-16 小时之间');

  /* Imax */
  if (isNaN(params.Imax))          errors.push('PAR 峰值 (Imax) 必须为数字');
  else if (params.Imax < 500 || params.Imax > 2500) errors.push('PAR 峰值 (Imax) 应在 500-2500 之间');

  /* RH */
  if (isNaN(params.RH))            errors.push('日均湿度 (RH) 必须为数字');
  else if (params.RH < 10 || params.RH > 100)        errors.push('日均湿度 (RH) 应在 10-100% 之间');

  /* k */
  if (isNaN(params.k))             errors.push('降温系数 (k) 必须为数字');
  else if (params.k < 5 || params.k > 25)           errors.push('降温系数 (k) 应在 5-25 之间');

  /* T0 */
  if (isNaN(params.T0))            errors.push('HHA 高温阈值 (T0) 必须为数字');
  else if (params.T0 < 30 || params.T0 > 45)        errors.push('HHA 高温阈值 (T0) 应在 30-45°C 之间');

  /* 表格模式最低行数 */
  if (params.useTable && params.productTable.length < 2)
    errors.push('表格模式下产品数据至少需要 2 行');

  /* 产品表物理趋势校验（遮阳率递增=浓度变浓） */
  if (params.useTable && params.productTableSorted && params.productTableSorted.length >= 2) {
    const sorted = params.productTableSorted;
    for (let i = 0; i < sorted.length - 1; i++) {
      const curr = sorted[i];
      const next = sorted[i + 1];
      if (curr.tempDrop > next.tempDrop) {
        errors.push('产品数据物理冲突：遮阳率增加(变浓)时，降温量不能下降。请检查输入。');
        break;
      }
      if (curr.ratioN < next.ratioN) {
        errors.push('产品数据物理冲突：遮阳率增加(变浓)时，兑水比应减小(更浓)。请检查输入。');
        break;
      }
      if (curr.coverage < next.coverage) {
        errors.push('产品数据物理冲突：遮阳率增加(变浓)时，每kg喷洒面积应减小。请检查输入。');
        break;
      }
    }
  }

  /* 大棚面积 */
  if (isNaN(params.sprayArea) || params.sprayArea <= 0 || params.sprayArea > 500)
    errors.push('大棚面积应在 0.1-500 亩之间');

  /* 喷雾器容量 */
  if (isNaN(params.sprayerCap) || params.sprayerCap < 1 || params.sprayerCap > 2000)
    errors.push('喷雾器容量应在 1-2000 L 之间');

  /* 粉剂单价 */
  if (isNaN(params.powderPrice) || params.powderPrice < 0 || params.powderPrice > 10000)
    errors.push('粉剂单价应在 0-10000 元/kg 之间');

  return { valid: errors.length === 0, errors };
}

/* ---- 降温量获取（表格插值 vs k*R） ---- */
function getTempDrop(R, useTable, productTable, k) {
  const tbl = productTable;
  if (useTable && tbl && tbl.length >= 1) {
    return Math.max(0, linearInterpolate(tbl, 'reflectance', 'tempDrop', R));
  }
  return k * R;
}

/* ---- 主入口 ---- */
function findOptimalR(params) {
  const {
    LSP, LCP, Tmax, Tmin, D, Imax, k, T0, useTable, RH,
    spectrumMode, spectrumControlPoints
  } = params;
  // 优先使用排序后的 productTable，回退到原始
  const productTable = params.productTableSorted || params.productTable;

  const baseTemps = generateHourlyTemps(Tmax, Tmin, D);
  const baseA = calcPhotosynth(0, LSP, LCP, Imax, D);
  if (baseA === 0) {
    return { results: [], optimum: null, error: '基准光合量为零，请检查 LSP/LCP/Imax 参数' };
  }

  /* 光谱模式预计算：归一化基础波形 → 积分得单位振幅常数 → 循环内 O(1) */
  let K_solar = null;  // S_total(R) = R × K_solar
  let K_par   = null;  // Rpar_loss(R) = R × K_par
  if (spectrumMode && spectrumControlPoints && spectrumControlPoints.length >= 2) {
    const baseCurve = fritschCarlsonSpline(spectrumControlPoints, 380, 1100, 10);
    const peak = Math.max(...baseCurve.map(p => p.y), 0.001);
    const normCurve = baseCurve.map(p => ({ x: p.x, y: p.y / peak }));
    const { S_total: Sn, Tau_PAR_eff: Te } = integrateSpectrum(normCurve);
    K_solar = Sn;      // 单位振幅下的总遮阳率
    K_par   = 1 - Te;  // 单位振幅下的 PAR 损失率
  }

  let bestIndex = -1;
  let bestY     = -Infinity;
  const results = [];
  const cooledBuf = new Array(baseTemps.length);  // 复用缓冲区，避免每次 map 创建新数组

  const steps = Math.round((CONFIG.R_MAX - CONFIG.R_MIN) / CONFIG.R_STEP);
  for (let i = 0; i <= steps; i++) {
    const R = CONFIG.R_MIN + i * CONFIG.R_STEP;
    let deltaT, A_rel;

    if (K_solar != null) {
      // 光谱模式：O(1) 乘法替代循环内样条+积分
      const S_total = R * K_solar;
      const Tau_PAR_eff = 1 - R * K_par;
      deltaT = getTempDrop(S_total, useTable, productTable, k);
      A_rel = photosynthCore(Tau_PAR_eff, LSP, LCP, Imax, D) / baseA;
    } else {
      deltaT = getTempDrop(R, useTable, productTable, k);
      A_rel = calcPhotosynth(R, LSP, LCP, Imax, D) / baseA;
    }

    for (let j = 0; j < baseTemps.length; j++) cooledBuf[j] = baseTemps[j] - deltaT;
    const HHA = calcHHA(cooledBuf, T0, RH);
    const L   = calcLossRate(HHA);
    const Y   = Math.max(0, Math.min(1, A_rel * (1 - L)));

    results.push({
      R:            Math.round(R * 100) / 100,
      Y,
      A_rel,
      L,
      HHA:          Math.round(HHA * 10) / 10,
      Tmax_cooled:  +(Tmax - deltaT).toFixed(1),
      deltaT:       +deltaT.toFixed(1),
      S_total:      K_solar != null ? +(R * K_solar).toFixed(3) : null
    });

    if (Y > bestY) {
      bestY     = Y;
      bestIndex = results.length - 1;
    }
  }

  /* 近优平台区间 */
  let plateauRMin = null;
  let plateauRMax = null;
  if (bestIndex >= 0 && bestY > -Infinity) {
    const near = results.filter(r => r.Y >= bestY * (1 - CONFIG.PLATEAU_EPSILON));
    if (near.length > 0) {
      plateauRMin = Math.min(...near.map(r => r.R));
      plateauRMax = Math.max(...near.map(r => r.R));
    }
  }

  let optimum = null;
  if (plateauRMin != null) {
    optimum = results.find(r => Math.abs(r.R - plateauRMin) < 1e-6) || results[bestIndex];
  } else if (bestIndex >= 0) {
    optimum = results[bestIndex];
  }

  return { results, optimum, plateauRMin, plateauRMax, bestIndex, error: null };
}

/**
 * 计算喷雾器分批信息
 * @param {number} powderKg  总粉量 (kg)
 * @param {number} waterL    总水量 (L)
 * @param {number} sprayerCap 喷雾器容量 (L)
 * @param {boolean} isEstimate 是否为估算值
 */
function getSprayerBatches(powderKg, waterL, sprayerCap, isEstimate) {
  if (!sprayerCap || sprayerCap <= 0) {
    return { batches: 0, powderPerBatch: 0, waterPerBatch: 0, safeCap: 0, isEstimate };
  }
  const safeCap = sprayerCap * CONFIG.SPRAYER_SAFETY_FACTOR;
  const batches = Math.ceil(waterL / safeCap);
  return {
    batches,
    powderPerBatch: powderKg / batches,
    waterPerBatch:  waterL / batches,
    safeCap,
    isEstimate
  };
}

/** 表格模式建议
 * @param {number} sTotalOpt — 光谱模式的 S_total，非 null 时替代 Ropt 查表
 */
function getTableAdvice(Ropt, productTable, sprayAreaMu, sTotalOpt = null) {
  if (!productTable || productTable.length < 2) {
    return {
      value: '产品数据至少需要 2 行',
      status: 'insufficient',
      detail: null
    };
  }

  const refs = productTable.map(r => r.reflectance);
  const rMin = Math.min(...refs);
  const rMax = Math.max(...refs);

  // 光谱模式：用 S_total 替代 Ropt 查表，并校验范围
  const lookupVal = (sTotalOpt != null) ? sTotalOpt : Ropt;

  if (lookupVal < rMin || lookupVal > rMax) {
    const p = v => (v * 100).toFixed(0) + '%';
    if (sTotalOpt != null) {
      return {
        value: `该光谱曲线算出的总遮阳率 S_total=${p(sTotalOpt)}，已超出当前产品数据的有效范围 [${p(rMin)}–${p(rMax)}]，请调整光谱曲线或补充产品数据`,
        status: 'out_of_range',
        detail: null
      };
    }
    return {
      value: `R_opt 超出产品数据范围 [${p(rMin)}–${p(rMax)}]，请扩展`,
      status: 'out_of_range',
      detail: null
    };
  }

  const ratioStr   = nearestStr(productTable, 'reflectance', 'ratio', lookupVal);
  const coverage   = linearInterpolate(productTable, 'reflectance', 'coverage', lookupVal);
  const totalM2    = (sprayAreaMu || 1) * CONFIG.MU_TO_M2;
  const powderKg   = totalM2 / coverage;
  const waterRatio = parseFloat(ratioStr.split(':')[1]) || 8;
  const waterL     = powderKg * waterRatio;

  return {
    value: [
      `兑水比 ${ratioStr}`,
      `每 kg 粉喷 ${coverage.toFixed(0)} ㎡`,
      `共需 ${powderKg.toFixed(1)} kg 粉 + ${waterL.toFixed(0)} L 水`
    ].join('  ·  '),
    status: 'ok',
    detail: {
      ratio:    ratioStr,
      coverage: `${coverage.toFixed(0)} ㎡/kg`,
      powder:   String(powderKg),
      water:    String(waterL),
      powderKg,
      waterL,
      isEstimate: false
    }
  };
}

/**
 * K 值模式稀释比建议（含估算 detail）
 * 基于 standardRatio → coverage 经验映射估算
 */
function interpolateDilution(Ropt, mapping, sprayAreaMu) {
  const muToM2 = (sprayAreaMu || 1) * CONFIG.MU_TO_M2;

  if (!mapping || mapping.length === 0) {
    return { value: '请填写稀释比映射表', status: 'empty', mode: 'k', detail: null };
  }
  if (mapping.length < 2) {
    return { value: '请至少填写两组映射数据', status: 'insufficient', mode: 'k', detail: null };
  }

  const sorted = [...mapping].sort((a, b) => a.r - b.r);

  if (Ropt < sorted[0].r || Ropt > sorted[sorted.length - 1].r) {
    const p = v => (v * 100).toFixed(0) + '%';
    return {
      value: `R_opt 超出范围 [${p(sorted[0].r)}–${p(sorted[sorted.length - 1].r)}]`,
      status: 'out_of_range',
      mode: 'k',
      detail: null
    };
  }

  let lo = sorted[0];
  let hi = sorted[sorted.length - 1];
  for (let i = 0; i < sorted.length - 1; i++) {
    if (Ropt >= sorted[i].r && Ropt <= sorted[i + 1].r) {
      lo = sorted[i];
      hi = sorted[i + 1];
      break;
    }
  }

  const ratio = (Math.abs(hi.r - lo.r) < 1e-9)
    ? lo.ratio
    : lo.ratio + (Ropt - lo.r) / (hi.r - lo.r) * (hi.ratio - lo.ratio);

  // 基于 CONFIG.K_MODE_COVERAGE_MAP 二分插值估算 coverage
  const ratioCovg = CONFIG.K_MODE_COVERAGE_MAP;
  const ri = Math.max(1, Math.min(20, ratio));
  const estCoverage = Math.max(50, Math.min(5000, linearInterpolate(ratioCovg, 'r', 'c', ri)));

  const powderKg = muToM2 / estCoverage;
  const waterL   = powderKg * ratio;

  return {
    value: `建议兑水比例 1:${ratio.toFixed(1)}`,
    status: 'ok',
    mode: 'k',
    detail: {
      ratio:    `1:${ratio.toFixed(1)}`,
      coverage: `${estCoverage.toFixed(0)} ㎡/kg (估算)`,
      powder:   String(powderKg),
      water:    String(waterL),
      powderKg,
      waterL,
      isEstimate: true
    }
  };
}

/** 安全省钱：近优平台 1/4 分位，若无平台则回退 bestIndex */
function safeSavePoint(results, bestIndex, plateau) {
  if (plateau && plateau.rMin != null && plateau.rMax != null) {
    const rQuarter = plateau.rMin + (plateau.rMax - plateau.rMin) * CONFIG.PLATEAU_QUARTER;
    let nearestIdx = bestIndex, minDiff = Infinity;
    for (let i = 0; i < results.length; i++) {
      const diff = Math.abs(results[i].R - rQuarter);
      if (diff < minDiff) { minDiff = diff; nearestIdx = i; }
    }
    return results[nearestIdx];
  }
  return results[bestIndex];
}

/**
 * @param {Array} results — findOptimalR 返回的 results 数组
 * @param {number} bestIndex — 最高 Y 的索引
 * @param {object} plateau — { rMin, rMax } 或 null
 * @param {object} params — 用户输入参数（含 T0）
 * @param {object} enabledFlags — { safe_save: true, max_y: boolean, absolute_temp: boolean }
 * @returns {{ points: Array, warnings: Array }}
 */
function extractStrategies(results, bestIndex, plateau, params, enabledFlags) {
  const points = [];
  const warnings = [];
  if (!results || results.length === 0) return { points, warnings };

  // 安全省钱始终生成
  const baseData = safeSavePoint(results, bestIndex, plateau);
  points.push(makeStrategyPoint('safe_save', STRATEGY_DEFS.safe_save, baseData, 'preset'));

  if (enabledFlags.max_y) {
    const maxPt = results[bestIndex];
    points.push(makeStrategyPoint('max_y', STRATEGY_DEFS.max_y, maxPt, 'preset'));
  }

  if (enabledFlags.absolute_temp) {
    const safeResults = results.filter(r => r.Tmax_cooled <= params.T0 + 0.05);
    if (safeResults.length === 0) {
      warnings.push({ level: 'warning',
        text: '当前气候极端，降温剂无法保证棚温降至阈值以下，建议结合遮阳网或通风措施' });
    } else {
      // 取温度安全范围内综合效益最优（而非最小R卡边，天气微变就破防）
      const bestInSafe = safeResults.reduce((a, b) => a.Y > b.Y ? a : b);
      points.push(makeStrategyPoint('absolute_temp', STRATEGY_DEFS.absolute_temp, bestInSafe, 'preset'));
    }
  }

  return { points, warnings };
}

function evaluatePointWarnings(point, params) {
  const warnings = [];
  const { data } = point;
  if (data.Tmax_cooled > params.T0) {
    warnings.push({ level: 'warning',
      text: `降温后棚温仍达 ${data.Tmax_cooled}°C，超过 HHA 阈值 (${params.T0}°C)，存在热害风险` });
  }
  if (data.A_rel < 0.60) {
    warnings.push({ level: 'warning',
      text: `光合保留率仅 ${(data.A_rel * 100).toFixed(0)}%，过度遮阳可能严重影响产量` });
  }
  if (data.L > 0.20) {
    warnings.push({ level: 'warning',
      text: `理论产量损失率达 ${(data.L * 100).toFixed(0)}%，建议结合物理降温措施` });
  }
  if (data.R >= CONFIG.R_MAX - 0.005) {
    warnings.push({ level: 'info',
      text: '该策略遮阳率已达系统上限，可能仍不满足需求' });
  }
  return warnings;
}

function evaluateEnvWarnings(params) {
  const env = [];
  if (params.Tmax > 50) {
    env.push({ level: 'warning',
      text: `棚内日最高温 ${params.Tmax}°C 属极端高温，植物生理机能可能已受不可逆损伤` });
  }
  if (params.RH > 85) {
    env.push({ level: 'warning',
      text: `棚内日间均湿 ${params.RH}% 过高，阻碍蒸腾，建议加强通风` });
  }
  if (params.Tmin < 10) {
    env.push({ level: 'info',
      text: `夜间最低温 ${params.Tmin}°C 偏低，需关注冷害风险` });
  }
  return env;
}

/** 主渲染函数 */
function renderChart(results, optimum, activeStrategies, onCustomPointAdd, onPresetPointClick, plateau) {
  const ctx = document.getElementById('yieldChart').getContext('2d');
  const ph  = document.getElementById('chart-placeholder');

  if (!results || !results.length) {
    ph.style.display = 'flex';
    if (chartInstance) { chartInstance.destroy(); chartInstance = null; }
    return;
  }

  ph.style.display = 'none';

  const hasPlateau = plateau && plateau.rMin != null && plateau.rMax != null
    && Math.abs(plateau.rMax - plateau.rMin) > 0.005;

  const labels  = results.map(r => (r.R * 100).toFixed(0) + '%');
  const yData   = results.map(r => r.Y);
  const pColors = results.map(() => 'rgba(45,106,79,0.3)');
  const pSizes  = results.map(() => 3);

  if (chartInstance) {
    chartInstance.data.labels = labels;
    chartInstance.data.datasets[0].data = yData;
    chartInstance.data.datasets[0].pointBackgroundColor = pColors;
    chartInstance.data.datasets[0].pointRadius = pSizes;
    chartInstance.optPluginData = { results, optimum, plateau, hasPlateau, activeStrategies, onCustomPointAdd, onPresetPointClick };
    chartInstance.update('none');
  } else {
    chartInstance = new Chart(ctx, {
      type: 'line',
      data: {
        labels,
        datasets: [{
          label: '综合效益 Y', data: yData,
          borderColor: '#2d6a4f', backgroundColor: 'rgba(45,106,79,0.06)',
          borderWidth: 2.5, fill: true, tension: 0.35,
          pointBackgroundColor: pColors, pointBorderColor: pColors,
          pointRadius: pSizes, pointHoverRadius: 7, pointBorderWidth: 1
        }]
      },
      options: {
        responsive: true, maintainAspectRatio: false,
        onClick: (event, elements, chart) => {
          if (elements.length > 0) {
            const index = elements[0].index;
            const r = chart.optPluginData.results[index];
            if (!r) return;
            const strategies = chart.optPluginData.activeStrategies || [];
            // 命中已有自定义点 → 删除
            const existingCustom = strategies.find(s =>
              s.type === 'custom' && Math.abs(s.data.R - r.R) < 0.005);
            if (existingCustom) {
              if (chart.optPluginData.onCustomPointDelete) {
                chart.optPluginData.onCustomPointDelete(existingCustom.id);
              }
              return;
            }
            // 命中预设点 → Toast
            if (strategies.some(s => s.type === 'preset' && Math.abs(s.data.R - r.R) < 0.005)) {
              if (chart.optPluginData.onPresetPointClick) {
                chart.optPluginData.onPresetPointClick();
              }
              return;
            }
            // 正常新增
            if (chart.optPluginData.onCustomPointAdd) {
              chart.optPluginData.onCustomPointAdd(r);
            }
          }
        },
        interaction: { intersect: false, mode: 'index' },
        plugins: {
          legend: { display: true, labels: { usePointStyle: true, padding: 20 } },
          tooltip: { callbacks: { label(ctx) {
            const r = ctx.chart.optPluginData.results[ctx.dataIndex];
            return [`Y:${r.Y.toFixed(4)}`,`A_rel:${(r.A_rel*100).toFixed(1)}%`,`L:${(r.L*100).toFixed(1)}%`,`HHA:${r.HHA.toFixed(1)}°C·h`,`棚温:${r.Tmax_cooled}°C`];

        scales: {
          x: { title: { display: true, text: '遮阳率 R', font: { weight: 'bold' } }, grid: { color: '#eee' } },
          y: { title: { display: true, text: '综合效益 Y', font: { weight: 'bold' } }, min: 0, max: 1, grid: { color: '#eee' } }
        }
      },
      plugins: [optAnnotationPlugin()]
    });
    chartInstance.optPluginData = { results, optimum, plateau, hasPlateau, activeStrategies, onCustomPointAdd, onPresetPointClick };
  }
}

function optAnnotationPlugin() {
  return {
    id: 'optAnno',
    beforeDatasetsDraw(chart) {
      const { results, plateau, hasPlateau } = chart.optPluginData || {};
      if (!hasPlateau) return;
      const meta = chart.getDatasetMeta(0);
      if (!meta.data || !meta.data.length) return;
      const iMin = results.findIndex(r => Math.abs(r.R - plateau.rMin) < 0.005);
      const iMax = results.findIndex(r => Math.abs(r.R - plateau.rMax) < 0.005);
      if (iMin < 0 || iMax < 0) return;
      const { ctx, chartArea } = chart;
      ctx.save();
      ctx.fillStyle = 'rgba(255, 193, 7, 0.12)';
      ctx.fillRect(Math.min(meta.data[iMin].x, meta.data[iMax].x), chartArea.top,
        Math.abs(meta.data[iMax].x - meta.data[iMin].x), chartArea.bottom - chartArea.top);
      ctx.restore();
    },
    afterDraw(chart) {
      const { ctx, chartArea } = chart;
      const { results, activeStrategies } = chart.optPluginData || {};
      const meta = chart.getDatasetMeta(0);
      if (!meta.data || !meta.data.length) return;

      if (!activeStrategies || !activeStrategies.length) return;

      // 按 R 分组（整数键）
      const groups = {};
      activeStrategies.forEach(s => {
        const key = Math.round(s.data.R * 200);
        if (!groups[key]) groups[key] = [];
        groups[key].push(s);
      });

      Object.values(groups).forEach(group => {
        const first = group[0];
        const idx = results.findIndex(r => Math.abs(r.R - first.data.R) < 0.005);
        if (idx < 0) return;
        const pt = meta.data[idx];
        if (!pt) return;

        // 垂直虚线（save/restore 防污染）
        ctx.save();
        ctx.setLineDash([4, 4]);
        ctx.strokeStyle = first.color + '80';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(pt.x, chartArea.top);
        ctx.lineTo(pt.x, pt.y);
        ctx.stroke();
        ctx.restore();

        const radius = 6;
        const sliceAngle = (Math.PI * 2) / group.length;
        const startOffset = -Math.PI / 2;

        if (group.length === 1) {
          // 单点：实心圆
          ctx.save();
          ctx.fillStyle = first.color;
          ctx.beginPath();
          ctx.arc(pt.x, pt.y, radius, 0, Math.PI * 2);
          ctx.fill();
          ctx.strokeStyle = '#fff'; ctx.lineWidth = 1;
          ctx.stroke();
          ctx.restore();
        } else {
          // 重合点：扇区 + 放射短文本
          group.forEach((s, i) => {
            const a0 = startOffset + i * sliceAngle;
            const a1 = a0 + sliceAngle;

            ctx.beginPath();
            ctx.moveTo(pt.x, pt.y);
            ctx.arc(pt.x, pt.y, radius, a0, a1);
            ctx.closePath();
            ctx.fillStyle = s.color; ctx.fill();
            ctx.strokeStyle = '#fff'; ctx.lineWidth = 2; ctx.stroke();
          });
        }
      });
    }
  };
}

/** 构建 Diff Grid 对比表 */
function updateDiffTable(activeStrategies, baseStrategyId, onSetBase, onDeleteCustom, powderPrice) {
  const container = document.getElementById('diff-table-container');
  const detailCard = document.getElementById('detail-card');
  if (!container || !detailCard) return;

  if (!activeStrategies || activeStrategies.length === 0) {
    detailCard.style.display = 'none'; return;
  }
  detailCard.style.display = 'block';

  const base = activeStrategies.find(s => s.id === baseStrategyId) || activeStrategies[0];
  const N = activeStrategies.length;

  let html = `<div class="diff-grid" style="grid-template-columns: 140px repeat(${N}, 1fr);">`;

  // Header row
  html += '<div class="diff-header">指标</div>';
  activeStrategies.forEach(sp => {
    const isBase = sp.id === baseStrategyId;
    html += `<div class="diff-header" style="color:${sp.color}">${sp.name}`;
    if (sp.type === 'custom' && sp.id !== 'safe_save' && onDeleteCustom) {
      html += ` <button data-action="delete" data-id="${sp.id}" style="border:none;background:none;color:#c0392b;cursor:pointer;font-size:1rem;">✕</button>`;
    }
    if (!isBase && onSetBase) {
      html += ` <a data-action="set-base" data-id="${sp.id}" style="color:#1976d2;cursor:pointer;font-size:0.75rem;">设为基准</a>`;
    }
    html += '</div>';
  });

  // Data rows
  METRICS_META.forEach(meta => {
    const baseVal = base.data[meta.key];
    html += `<div class="diff-label">${meta.label}</div>`;
    activeStrategies.forEach(sp => {
      const isBase = sp.id === baseStrategyId;
      const val = sp.data[meta.key];
      if (isBase) {
        html += `<div class="diff-base">${meta.fmt(val)}</div>`;
      } else {
        html += `<div class="diff-cell">${meta.fmt(val)} ${renderDiffBadge(baseVal, val, meta.higherIsBetter)}</div>`;
      }
    });
  });

  // Advice rows (from sp.advice.detail)
  html += '<div class="diff-label">兑水比</div>';
  activeStrategies.forEach(sp => {
    const isBase = sp.id === baseStrategyId;
    const ratio = sp.advice?.detail?.ratio || '--';
    if (isBase) html += `<div class="diff-base">${ratio}</div>`;
    else html += `<div class="diff-cell">${ratio}</div>`;
  });

  html += '<div class="diff-label">总粉剂用量</div>';
  activeStrategies.forEach(sp => {
    const isBase = sp.id === baseStrategyId;
    const kg = sp.advice?.detail?.powderKg;
    const val = (kg != null) ? parseFloat(kg).toFixed(1) + ' kg' : '--';
    if (isBase) html += `<div class="diff-base">${val}</div>`;
    else {
      const baseKg = base.advice?.detail?.powderKg;
      const diff = (kg != null && baseKg != null) ? renderDiffBadge(parseFloat(baseKg), parseFloat(kg), false) : '';
      html += `<div class="diff-cell">${val} ${diff}</div>`;
    }
  });

  html += '<div class="diff-label">预计总成本</div>';
  activeStrategies.forEach(sp => {
    const isBase = sp.id === baseStrategyId;
    const cost = (sp.advice?.detail?.powderKg != null)
      ? '¥' + (parseFloat(sp.advice.detail.powderKg) * (Number.isFinite(powderPrice) ? powderPrice : 0)).toFixed(2)
      : '--';
    if (isBase) html += `<div class="diff-base">${cost}</div>`;
    else html += `<div class="diff-cell">${cost}</div>`;
  });

  html += '</div>';
  container.innerHTML = html;

  // Event delegation for actions
  container.querySelectorAll('[data-action="set-base"]').forEach(el => {
    el.addEventListener('click', (e) => {
      e.stopPropagation();
      if (onSetBase) onSetBase(el.dataset.id);
    });
  });
  container.querySelectorAll('[data-action="delete"]').forEach(el => {
    el.addEventListener('click', (e) => {
      e.stopPropagation();
      if (onDeleteCustom) onDeleteCustom(el.dataset.id);
    });
  });
}

function renderDiffBadge(baseVal, curVal, higherIsBetter) {
  if (higherIsBetter == null) return '';
  const diff = curVal - baseVal;
  if (Math.abs(diff) < 0.001) return '<span class="diff-same">--</span>';
  const isGood = (diff > 0 && higherIsBetter) || (diff < 0 && !higherIsBetter);
  const cls = isGood ? 'diff-good' : 'diff-bad';
  const sign = diff > 0 ? '+' : '';
  return `<span class="${cls}">${sign}${diff.toFixed(3)}</span>`;
}

        ok.textContent = '✓ 各项指标均处于安全区间';

    });
  }

function resizeCanvas() {
  if (!_canvas) return;
  const dpr = window.devicePixelRatio || 1;
  const rect = _canvas.parentElement.getBoundingClientRect();
  const w = rect.width || 700;
  const h = rect.height || 450;
  _canvas.width = w * dpr;
  _canvas.height = h * dpr;
  _canvas.style.width = w + 'px';
  _canvas.style.height = h + 'px';
  _ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
}

function render() {
  if (!_ctx || !_canvas) return;
  const ctx = _ctx;
  const W = _canvas.width / (window.devicePixelRatio || 1);
  const H = _canvas.height / (window.devicePixelRatio || 1);

  const pad = { top: 30, right: 50, bottom: 50, left: 50 };
  const pw = W - pad.left - pad.right;
  const ph = H - pad.top - pad.bottom;

  ctx.clearRect(0, 0, W, H);

  /* ---- 背景渐变（380-780 彩虹, 780-1100 暗红） ---- */
  const grad = ctx.createLinearGradient(pad.left, 0, pad.left + pw, 0);
  const a = _bgAlpha;
  grad.addColorStop(0, `rgba(138,43,226,${a})`);
  grad.addColorStop(0.15, `rgba(0,0,255,${a * 0.8})`);
  grad.addColorStop(0.3, `rgba(0,255,0,${a * 0.8})`);
  grad.addColorStop(0.42, `rgba(255,255,0,${a * 0.67})`);
  grad.addColorStop(0.56, `rgba(255,0,0,${a * 0.67})`);
  grad.addColorStop(0.57, `rgba(80,0,0,${a * 0.67})`);
  ctx.fillStyle = grad;
  ctx.fillRect(pad.left, pad.top, pw, ph);

  /* ---- 网格线 ---- */
  ctx.strokeStyle = '#e0e0e0';
  ctx.lineWidth = 0.5;
  for (let i = 0; i <= 5; i++) {
    const y = pad.top + ph * i / 5;
    ctx.beginPath(); ctx.moveTo(pad.left, y); ctx.lineTo(pad.left + pw, y); ctx.stroke();
    ctx.fillStyle = '#888'; ctx.font = '11px sans-serif'; ctx.textAlign = 'right';
    ctx.fillText((1 - i / 5).toFixed(1), pad.left - 8, y + 4);
  }

  /* ---- 左 Y 轴标题 ---- */
  ctx.save();
  ctx.fillStyle = '#555'; ctx.font = 'bold 12px sans-serif'; ctx.textAlign = 'center';
  ctx.translate(12, pad.top + ph / 2);
  ctx.rotate(-Math.PI / 2);
  ctx.fillText('遮阳率', 0, 0);
  ctx.restore();

  /* ---- X 轴标签 ---- */
  ctx.fillStyle = '#555'; ctx.font = '11px sans-serif'; ctx.textAlign = 'center';
  [380, 500, 600, 700, 800, 950, 1100].forEach(wl => {
    const x = pad.left + pw * (wl - WL_MIN) / (WL_MAX - WL_MIN);
    ctx.fillText(wl + 'nm', x, pad.top + ph + 20);
  });

  /* ---- 右 Y 轴标题 ---- */
  ctx.save();
  ctx.fillStyle = '#555'; ctx.font = 'bold 12px sans-serif'; ctx.textAlign = 'center';
  ctx.translate(W - 12, pad.top + ph / 2);
  ctx.rotate(-Math.PI / 2);
  ctx.fillText('植物光合响应指数 (无量纲)', 0, 0);
  ctx.restore();

  /* ---- 植物响应曲线（虚线） ---- */
  const plantCurve = _plantPoints.length >= 2
    ? fritschCarlsonSpline(_plantPoints, WL_MIN, WL_MAX, WL_STEP)
    : null;
  if (plantCurve) {
    ctx.save();
    ctx.setLineDash([6, 4]);
    ctx.strokeStyle = 'rgba(100,180,100,0.5)';
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    plantCurve.forEach((p, i) => {
      const x = pad.left + pw * (p.x - WL_MIN) / (WL_MAX - WL_MIN);
      const y = pad.top + ph * (1 - p.y);
      if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    });
    ctx.stroke();
    ctx.setLineDash([]);
    ctx.restore();
  }

  /* ---- 主曲线（遮阳率） ---- */
  const curve = _controlPoints.length >= 2
    ? fritschCarlsonSpline(_controlPoints, WL_MIN, WL_MAX, WL_STEP)
    : null;
  if (curve) {
    ctx.save();
    ctx.fillStyle = 'rgba(45,106,79,0.15)';
    ctx.strokeStyle = '#2d6a4f';
    ctx.lineWidth = 2;
    ctx.beginPath();
    curve.forEach((p, i) => {
      const x = pad.left + pw * (p.x - WL_MIN) / (WL_MAX - WL_MIN);
      const y = pad.top + ph * (1 - p.y);
      if (i === 0) { ctx.moveTo(x, y); } else { ctx.lineTo(x, y); }
    });
    ctx.lineTo(pad.left + pw * (WL_MAX - WL_MIN) / (WL_MAX - WL_MIN), pad.top + ph);
    ctx.lineTo(pad.left, pad.top + ph);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();
    ctx.restore();
  }

  /* ---- 780nm 分割线（可见/红外） ---- */
  const x780 = pad.left + pw * (780 - WL_MIN) / (WL_MAX - WL_MIN);
  ctx.save();
  ctx.strokeStyle = 'rgba(50,50,50,0.5)';
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(x780, pad.top);
  ctx.lineTo(x780, pad.top + ph);
  ctx.stroke();
  // 顶部三角
  ctx.fillStyle = 'rgba(50,50,50,0.7)';
  ctx.beginPath();
  ctx.moveTo(x780, pad.top);
  ctx.lineTo(x780 - 4, pad.top + 6);
  ctx.lineTo(x780 + 4, pad.top + 6);
  ctx.closePath();
  ctx.fill();
  // 底部标签
  ctx.fillStyle = '#555'; ctx.font = '10px sans-serif'; ctx.textAlign = 'center';
  ctx.fillText('可见光', pad.left + pw * (580 - WL_MIN) / (WL_MAX - WL_MIN), pad.top + ph + 14);
  ctx.fillText('红外',   pad.left + pw * (880 - WL_MIN) / (WL_MAX - WL_MIN), pad.top + ph + 14);
  ctx.restore();

  /* ---- 控制点手柄（遮阳率实心圆 / 植物响应空心方块） ---- */
  const pts = _editingPlant ? _plantPoints : _controlPoints;
  pts.forEach((p, i) => {
    const x = pad.left + pw * (p.x - WL_MIN) / (WL_MAX - WL_MIN);
    const y = pad.top + ph * (1 - p.y);
    if (_editingPlant) {
      // 空心方块
      ctx.strokeStyle = 'rgba(100,180,100,0.9)'; ctx.lineWidth = 2;
      ctx.strokeRect(x - 4, y - 4, 8, 8);
    } else {
      ctx.beginPath();
      ctx.arc(x, y, i === _dragIdx ? 7 : 5, 0, 2 * Math.PI);
      ctx.fillStyle = 'rgba(45,106,79,0.9)';
      ctx.fill();
      ctx.strokeStyle = '#fff'; ctx.lineWidth = 1.5;
      ctx.stroke();
    }
  });

  /* ---- 右上角摘要 ---- */
  const sum = integrateSpectrum(curve || []);
  ctx.fillStyle = '#1b4332'; ctx.font = 'bold 12px sans-serif'; ctx.textAlign = 'right';
  ctx.fillText(
    `总遮阳率 ${(sum.S_total * 100).toFixed(1)}%  |  有效光合透射 ${(sum.Tau_PAR_eff * 100).toFixed(1)}%`,
    pad.left + pw, pad.top + 14
  );

  /* ---- 背景透明度滑动条 ---- */
  const sx = pad.left, sy = pad.top + ph + 28, sw = 120, sh = 12;
  ctx.fillStyle = '#f0f0f0';
  ctx.fillRect(sx, sy, sw, sh);
  ctx.strokeStyle = '#aaa'; ctx.lineWidth = 0.5;
  ctx.strokeRect(sx, sy, sw, sh);
  const hx = sx + (_bgAlpha / 0.3) * sw;
  ctx.fillStyle = '#2d6a4f';
  ctx.fillRect(hx - 4, sy - 2, 8, sh + 4);
  ctx.fillStyle = '#888'; ctx.font = '10px sans-serif'; ctx.textAlign = 'left';
  ctx.fillText('背景透明度', sx + sw + 8, sy + sh - 2);

  /* 缩略图同步 */
  updateThumbnail(curve, sum);
}

function getPad() {
  const W = _canvas.width / (window.devicePixelRatio || 1);
  const H = _canvas.height / (window.devicePixelRatio || 1);
  return { left: 50, right: 50, top: 30, bottom: 50, W, H,
    pw: W - 100, ph: H - 80 };
}

function nmToPixel(nm) {
  const pad = getPad();
  return pad.left + pad.pw * (nm - WL_MIN) / (WL_MAX - WL_MIN);
}

function pixelToNm(px) {
  const pad = getPad();
  return WL_MIN + (px - pad.left) / pad.pw * (WL_MAX - WL_MIN);
}

function pixelToVal(py) {
  const pad = getPad();
  return Math.max(0, Math.min(1, 1 - (py - pad.top) / pad.ph));
}

function hitTest(px, py) {
  const pts = _editingPlant ? _plantPoints : _controlPoints;
  const pad = getPad();
  for (let i = pts.length - 1; i >= 0; i--) {
    const cx = pad.left + pad.pw * (pts[i].x - WL_MIN) / (WL_MAX - WL_MIN);
    const cy = pad.top + pad.ph * (1 - pts[i].y);
    const dist = Math.sqrt((px - cx) ** 2 + (py - cy) ** 2);
    if (dist < 10) return i;
  }
  return -1;
}

function onMouseDown(e) {
  if (!_canvas) return;
  const rect = _canvas.getBoundingClientRect();
  const px = e.clientX - rect.left;
  const py = e.clientY - rect.top;

  // ★ 滑条优先：先判断是否命中滑条区域
  const pad = getPad();
  const sliderY = pad.top + pad.ph + 28;
  if (px >= pad.left && px <= pad.left + 120 && py >= sliderY && py <= sliderY + 12) {
    _isDraggingSlider = true;
    _canvas.style.cursor = 'ew-resize';
    updateSliderFromMouse(px);
    return;
  }

  const pts = _editingPlant ? _plantPoints : _controlPoints;
  const hit = hitTest(px, py);
  if (hit >= 0) {
    _dragIdx = hit;
    _canvas.style.cursor = 'grabbing';
    return;
  }

  // 空白处添加新点
  const nm = Math.round(pixelToNm(px) / 10) * 10;
  const val = pixelToVal(py);

  let insertAt = pts.length;
  for (let i = 0; i < pts.length; i++) {
    if (pts[i].x > nm) { insertAt = i; break; }
  }
  if (insertAt > 0 && nm - pts[insertAt - 1].x < 0.5) return;
  if (insertAt < pts.length && pts[insertAt].x - nm < 0.5) return;

  pts.splice(insertAt, 0, { x: nm, y: val });
  _dragIdx = insertAt;
  _canvas.style.cursor = 'grabbing';
  render();
  if (_editingPlant) savePlantResponse();
}

function onMouseMove(e) {
  if (!_canvas) return;
  const rect = _canvas.getBoundingClientRect();
  const px = e.clientX - rect.left;
  const py = e.clientY - rect.top;
  const pad = getPad();
  const sliderY = pad.top + pad.ph + 28;

  if (_isDraggingSlider) {
    updateSliderFromMouse(px);
    return;
  }

  if (_dragIdx >= 0) {
    const pts = _editingPlant ? _plantPoints : _controlPoints;
    let nm = Math.round(pixelToNm(px) / 10) * 10;
    if (_dragIdx > 0) nm = Math.max(nm, pts[_dragIdx - 1].x + 0.5);
    if (_dragIdx < pts.length - 1) nm = Math.min(nm, pts[_dragIdx + 1].x - 0.5);
    nm = Math.max(WL_MIN, Math.min(WL_MAX, nm));
    pts[_dragIdx].x = nm;
    pts[_dragIdx].y = pixelToVal(py);
    render();
    if (_editingPlant) savePlantResponse();
    if (_onChange) _onChange();
  } else {
    // 滑条hover游标反馈
    if (px >= pad.left && px <= pad.left + 120 && py >= sliderY && py <= sliderY + 12) {
      _canvas.style.cursor = 'ew-resize';
    } else {
      const hit = hitTest(px, py);
      _canvas.style.cursor = hit >= 0 ? 'grab' : 'crosshair';
    }
  }
}

function onMouseUp() {
  if (_isDraggingSlider) { _isDraggingSlider = false; return; }
  _dragIdx = -1;
  if (_canvas) _canvas.style.cursor = 'crosshair';
}

function onDblClick(e) {
  if (!_canvas) return;
  const rect = _canvas.getBoundingClientRect();
  const px = e.clientX - rect.left;
  const py = e.clientY - rect.top;
  const pts = _editingPlant ? _plantPoints : _controlPoints;
  if (pts.length <= 2) return;
  const hit = hitTest(px, py);
  if (hit >= 0) {
    pts.splice(hit, 1);
    render();
    if (_editingPlant) savePlantResponse();
    if (_onChange) _onChange();
  }
}

function updateSliderFromMouse(px) {
  const pad = getPad();
  const sw = 120;
  _bgAlpha = Math.max(0, Math.min(0.3, ((px - pad.left) / sw) * 0.3));
  render();
}

let _boundMD = null, _boundMM = null, _boundMU = null, _boundDbl = null, _boundResize = null;

function bindCanvasEvents() {
  if (!_canvas) return;
  _boundMD = onMouseDown; _boundMM = onMouseMove;
  _boundMU = onMouseUp; _boundDbl = onDblClick;
  _canvas.addEventListener('mousedown', _boundMD);
  _canvas.addEventListener('mousemove', _boundMM);
  _canvas.addEventListener('mouseup', _boundMU);
  _canvas.addEventListener('dblclick', _boundDbl);
  _canvas.style.cursor = 'crosshair';
  _boundResize = resizeCanvas;
  window.addEventListener('resize', _boundResize);
}

function unbindCanvasEvents() {
  if (!_canvas) return;
  if (_boundMD) _canvas.removeEventListener('mousedown', _boundMD);
  if (_boundMM) _canvas.removeEventListener('mousemove', _boundMM);
  if (_boundMU) _canvas.removeEventListener('mouseup', _boundMU);
  if (_boundDbl) _canvas.removeEventListener('dblclick', _boundDbl);
  if (_boundResize) window.removeEventListener('resize', _boundResize);
}

function getVarietyKey() {
  const el = document.querySelector('.preset-btn.active');
  return el ? el.dataset.preset || 'custom' : 'custom';
}

function loadPlantResponse() {
  try {
    const raw = localStorage.getItem('gcc:plant:v1:' + getVarietyKey());
    if (raw) return JSON.parse(raw);
  } catch (e) { console.warn('[GCC] load plant response:', e); }
  const def = CONFIG.PLANT_RESPONSE_DEFAULT;
  const pts = []; let lastAdded = -100;
  for (let i = 0; i < def.length; i++) {
    const nm = WL_MIN + i * WL_STEP;
    if (i === 0 || i === def.length - 1 || def[i] > 0.1 && nm - lastAdded >= 80) {
      pts.push({ x: nm, y: def[i] });
      lastAdded = nm;
    }
  }
  if (!pts.find(p => p.x >= 700)) pts.push({ x: 780, y: 0 });
  if (!pts.find(p => p.x >= 1000)) pts.push({ x: 1100, y: 0 });
  return pts;
}

function savePlantResponse() {
  try {
    localStorage.setItem('gcc:plant:v1:' + getVarietyKey(), JSON.stringify(_plantPoints));
  } catch (e) { console.warn('[GCC] save plant response:', e); }
}

function resetPlantResponse() {
  _plantPoints = loadPlantResponse();
  render();
}

function updateThumbnail(curve, sum) {
  const thumb = document.getElementById('spectrum-thumb');
  if (!thumb) return;
  const tctx = thumb.getContext('2d');
  const tw = thumb.width, th = thumb.height;
  tctx.clearRect(0, 0, tw, th);

  if (curve) {
    tctx.fillStyle = 'rgba(45,106,79,0.2)';
    tctx.strokeStyle = '#2d6a4f';
    tctx.lineWidth = 1.5;
    tctx.beginPath();
    curve.forEach((p, i) => {
      const x = tw * (p.x - WL_MIN) / (WL_MAX - WL_MIN);
      const y = th * (1 - p.y);
      if (i === 0) tctx.moveTo(x, y); else tctx.lineTo(x, y);
    });
    tctx.lineTo(tw, th);
    tctx.lineTo(0, th);
    tctx.closePath();
    tctx.fill();
    tctx.stroke();
  }

  const st = document.getElementById('spectrum-s-total');
  const tp = document.getElementById('spectrum-tau-par');
  if (st && sum && !isNaN(sum.S_total)) st.textContent = (sum.S_total * 100).toFixed(1) + '%';
  if (tp && sum && !isNaN(sum.Tau_PAR_eff)) tp.textContent = (sum.Tau_PAR_eff * 100).toFixed(1) + '%';
}

const AppState = {
  dt: null,
  lastResult: null,
  lastAdvice: null,
  lastWarnings: null,
  lastParams: null,
  lastPlateau: null,
  activeStrategies: [],
  baseStrategyId: 'safe_save'
};
function scheduleAutoCalc() {
  clearTimeout(AppState.dt);
  AppState.dt = setTimeout(() => {
    try { saveState(collectPersistableState()); } catch (e) { console.warn('[GCC] saveState:', e); }
    doCalculation();
  }, CONFIG.DEBOUNCE_MS);
}

/* ---- 输出区状态复位（异常/校验失败时调用，杜绝残留旧数据） ---- */
function resetOutputUI(params) {
  renderChart(null, null, null, null, null, null);
  updateOutput(null, null, params.sprayArea, params.powderPrice, null);
  updateWarnings(null, null);
  setPrintEnabled(false);
}

/* ---- 计算流水线 ---- */
function doCalculation() {
  try {
    const params = collectParams();
    const v      = validateParams(params);
    highlightInvalidFields(params);

    if (!v.valid) {
      showToast(v.errors[0]);
      resetOutputUI(params);
      return;
    }

    // 创建排序副本供 linearInterpolate 二分查找使用
    if (params.useTable && params.productTable) {
      params.productTableSorted = [...params.productTable].sort((a, b) => a.reflectance - b.reflectance);
    }

    const btn  = document.getElementById('btn-calc');
    const orig = btn.textContent;
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>计算中...';

    setTimeout(() => {
      try {
        const result = findOptimalR(params);

        if (result.error) {
          showToast(result.error);
          resetOutputUI(params);
        } else {
        const plateau = (result.plateauRMin != null && result.plateauRMax != null)
          ? { rMin: result.plateauRMin, rMax: result.plateauRMax }
          : null;
        const bestIndex = result.bestIndex;

        // 策略提取
        const enabledFlags = {
          safe_save: true,
          max_y: document.getElementById('chk-max-y').checked,
          absolute_temp: document.getElementById('chk-absolute-temp').checked
        };
        const strat = extractStrategies(result.results, bestIndex >= 0 ? bestIndex : 0, plateau, params, enabledFlags);

        // 为每个策略点计算 advice
        const prodTbl = params.productTableSorted || params.productTable;
        strat.points.forEach(sp => {
          const lookupVal = (params.spectrumMode && sp.data.S_total != null)
            ? sp.data.S_total : sp.data.R;
          sp.advice = params.useTable
            ? getTableAdvice(lookupVal, prodTbl, params.sprayArea,
                params.spectrumMode && sp.data.S_total != null ? sp.data.S_total : null)
            : interpolateDilution(lookupVal, collectKMapping(), params.sprayArea);
          if (sp.advice && sp.advice.detail) {
            sp.advice.batchInfo = getSprayerBatches(
              sp.advice.detail.powderKg || 0,
              sp.advice.detail.waterL || 0,
              params.sprayerCap,
              sp.advice.detail.isEstimate
            );
          }
        });

        AppState.activeStrategies = strat.points;
        AppState.baseStrategyId = 'safe_save';

        // 策略点级 Warning + 环境 Warning
        const envWarnings = evaluateEnvWarnings(params);
        AppState.activeStrategies.forEach(point => {
          point.warnings = evaluatePointWarnings(point, params);
        });
        if (strat.warnings.length > 0) {
          strat.warnings.forEach(w => showToast(w.text));
        }

        const baseSp = strat.points.find(s => s.id === 'safe_save');
        const baseAdvice = baseSp ? baseSp.advice : null;

        renderChart(result.results, result.optimum, AppState.activeStrategies, onCustomPointAdd, onPresetPointClick, plateau);
        updateOptimumCard(result.optimum, baseAdvice, params.spectrumMode);
        updateDiffTable(AppState.activeStrategies, AppState.baseStrategyId, setBaseStrategy, deleteCustomStrategy, params.powderPrice);
        updateWarnings(envWarnings, AppState.activeStrategies);

        /* 存储供处方单使用 */
        AppState.lastResult   = result;
        AppState.lastAdvice   = baseAdvice;
        AppState.lastWarnings = envWarnings;
        AppState.lastParams   = { ...params, varietyName: getActiveVarietyName() };
        AppState.lastPlateau  = plateau;
        setPrintEnabled(true);
        }

        btn.disabled = false;
        btn.innerHTML = orig;
      } catch (innerErr) {
        console.error(innerErr);
        showToast('计算异常: ' + innerErr.message);
        resetOutputUI(params);
      } finally {
        btn.disabled = false;
        btn.innerHTML = orig;
      }
    }, 50);

  } catch (err) {
    console.error(err);
    showToast('计算异常: ' + err.message);
    try { resetOutputUI(collectParams()); } catch (_) {}
    const btn = document.getElementById('btn-calc');
    btn.disabled = false;
    btn.textContent = '▶ 计算';
  }
}

function getActiveVarietyName() {
  const active = document.querySelector('.preset-btn.active');
  if (!active) return '自定义';
  const preset = PRESETS[active.dataset.preset];
  return preset ? preset.name : '自定义';
}

function setPrintEnabled(enabled) {
  const btn = document.getElementById('btn-print');
  if (btn) btn.disabled = !enabled;
}

/* ---- 自定义点 + 基准切换回调 ---- */
function onCustomPointAdd(data) {
  // 替换模式：先删除已有自定义点（若有），保证只有一个参考点
  const oldIdx = AppState.activeStrategies.findIndex(s => s.type === 'custom');
  if (oldIdx >= 0) {
    if (AppState.baseStrategyId === AppState.activeStrategies[oldIdx].id) {
      AppState.baseStrategyId = 'safe_save';
    }
    AppState.activeStrategies.splice(oldIdx, 1);
  }

  const id = 'custom_1';
  const sp = {
    id, name: '参考点', color: '#e67e22',
    type: 'custom', isBase: false,
    data: { R: data.R, Y: data.Y, A_rel: data.A_rel, L: data.L,
      HHA: data.HHA, Tmax_cooled: data.Tmax_cooled, deltaT: data.deltaT,
      S_total: data.S_total || null },
    advice: null
  };
  // 计算 advice
  const params = AppState.lastParams || {};
  const prodTbl = params.productTableSorted || params.productTable;
  const lookupVal = (params.spectrumMode && sp.data.S_total != null) ? sp.data.S_total : sp.data.R;
  sp.advice = params.useTable
    ? getTableAdvice(lookupVal, prodTbl, params.sprayArea, params.spectrumMode && sp.data.S_total != null ? sp.data.S_total : null)
    : interpolateDilution(lookupVal, collectKMapping(), params.sprayArea);
  if (sp.advice?.detail) {
    sp.advice.batchInfo = getSprayerBatches(sp.advice.detail.powderKg || 0, sp.advice.detail.waterL || 0, params.sprayerCap, sp.advice.detail.isEstimate);
  }

  AppState.activeStrategies.push(sp);
  updateDiffTable(AppState.activeStrategies, AppState.baseStrategyId, setBaseStrategy, deleteCustomStrategy, AppState.lastParams?.powderPrice);
  if (AppState.lastResult) {
    renderChart(AppState.lastResult.results, AppState.lastResult.optimum, AppState.activeStrategies, onCustomPointAdd,
      onPresetPointClick,
      AppState.lastPlateau);
  }
}

function setBaseStrategy(id) {
  AppState.baseStrategyId = id;
  const baseSp = AppState.activeStrategies.find(s => s.id === id);
  if (baseSp) {
    updateOptimumCard(baseSp.data, baseSp.advice || null, AppState.lastParams?.spectrumMode);
  }
  updateDiffTable(AppState.activeStrategies, AppState.baseStrategyId, setBaseStrategy, deleteCustomStrategy, AppState.lastParams?.powderPrice);
}

function deleteCustomStrategy(id) {
  const idx = AppState.activeStrategies.findIndex(s => s.id === id);
  if (idx < 0) return;
  if (id === AppState.baseStrategyId) { AppState.baseStrategyId = 'safe_save'; }
  AppState.activeStrategies.splice(idx, 1);
  updateDiffTable(AppState.activeStrategies, AppState.baseStrategyId, setBaseStrategy, deleteCustomStrategy, AppState.lastParams?.powderPrice);
  if (AppState.lastResult) {
    renderChart(AppState.lastResult.results, AppState.lastResult.optimum, AppState.activeStrategies, onCustomPointAdd,
      onPresetPointClick,
      AppState.lastPlateau);
  }
}

function onCustomPointDelete(id) {
  deleteCustomStrategy(id);
}

/* ---- 从 localStorage 回填上次输入（刷新不丢失） ---- */
function restoreFromState() {
  const s = loadState();
  if (!s) return false;

  const numIds = ['LSP','LCP','T0','Tmax','Tmin','D','Imax','RH','k','k-visible','sprayArea','sprayerCap','powderPrice'];
  numIds.forEach(id => {
    const el = document.getElementById(id);
    if (el && s[id] != null && !isNaN(s[id])) el.value = s[id];
  });

  /* 模式 */
  if (s.useTable === false) {
    document.getElementById('mode-k').checked = true;
    document.getElementById('table-mode-panel').style.display = 'none';
    document.getElementById('k-mode-panel').style.display     = 'block';
  }

  /* 品种预设 */
  if (s.preset && s.preset !== 'custom') {
    const btn = document.querySelector(`[data-preset="${s.preset}"]`);
    if (btn) {
      document.querySelectorAll('.preset-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
    }
  } else if (s.preset === 'custom') {
    document.querySelectorAll('.preset-btn').forEach(b => b.classList.remove('active'));
    document.querySelector('[data-preset="custom"]').classList.add('active');
  }

  /* 产品表 / k 映射表 */
  if (Array.isArray(s.productTable) && s.productTable.length >= 2) {
    const tbody = document.getElementById('product-tbody');
    tbody.innerHTML = '';
    s.productTable.forEach(d => tbody.appendChild(createProductRow(d, scheduleAutoCalc)));
  }
  if (Array.isArray(s.kMapping) && s.kMapping.length >= 1) {
    const tbody = document.getElementById('k-mapping-tbody');
    tbody.innerHTML = '';
    s.kMapping.forEach(m => tbody.appendChild(createKMappingRow(m.r, m.ratio, scheduleAutoCalc)));
  }

  /* 光谱子模式 */
  if (s.spectrumSubMode === 'spectral') {
    const radio = document.getElementById('sub-spectral');
    if (radio) radio.checked = true;
    const panel = document.getElementById('spectrum-mode-panel');
    if (panel) panel.style.display = 'block';
  }
  if (s.spectrumControlPoints) {
    const el = document.getElementById('spectrum-control-points');
    if (el) el.value = typeof s.spectrumControlPoints === 'string' ? s.spectrumControlPoints : JSON.stringify(s.spectrumControlPoints);
  }

  return true;
}

/** 采集含预设标识与表格的完整状态用于持久化 */
function collectPersistableState() {
  const params = collectParams();
  const active = document.querySelector('.preset-btn.active');
  params.preset = active ? active.dataset.preset : 'custom';
  params.kMapping = collectKMapping();
  return params;
}

/* ---- 启动 ---- */
function init() {
  initProductTable(scheduleAutoCalc);
  initKMappingTable(scheduleAutoCalc);
  bindEvents();
  setPrintEnabled(false);
  const restored = restoreFromState();
  if (!restored) {
    /* 无存档时走默认计算（保持原行为） */
    doCalculation();
  } else {
    /* 触发一次保存 + 计算 */
    try { saveState(collectPersistableState()); } catch (e) { console.warn('[GCC] saveState init:', e); }
    doCalculation();
  }
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', init);
} else {
  init();
}


})();
